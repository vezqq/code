        function Utility:IsHovered(Position, Config)
            local ScreenPos, OnScreen = Camera:WorldToViewportPoint(Position)
            --
            if not OnScreen then
                return false
            end
            --
            local Delta = GetMouseLocation(InputService) - Vector2New(ScreenPos.X, ScreenPos.Y)
            local Width = Config.Width
            local Height = Config.Height
            local VP = Camera.ViewportSize
            local Left = (type(Width) == "table" and Width[1] or Width) * VP.X
            local Right = (type(Width) == "table" and Width[2] or Width) * VP.X
            local Up = (type(Height) == "table" and Height[1] or Height) * VP.Y
            local Down = (type(Height) == "table" and Height[2] or Height) * VP.Y
            return Delta.X >= -Left and Delta.X <= Right
                and Delta.Y >= -Up and Delta.Y <= Down
        end
    end

        function Character:GetVisible(Player)
            local GetCharacter = self:GetCharacters(Player)
            local RootPart = self:GetRootPart(Player)
            local LocalCharacter = self:GetCharacters(LocalPlayer)
            
            if not RootPart or not LocalCharacter then
                return false, nil
            end
            
            local Params = RaycastParams.new()
            Params.FilterType = Enum.RaycastFilterType.Exclude
            Params.FilterDescendantsInstances = { LocalCharacter }
            Params.IgnoreWater = true
            
            local Direction = (RootPart.Position - Camera.CFrame.Position)
            local Result = Workspace:Raycast(Camera.CFrame.Position, Direction, Params)
            
            if Result and Result.Instance then
                return Result.Instance:IsDescendantOf(GetCharacter), Result.Instance
            end
            
            return true, nil
        end

    local Closest = { }; do
        function Closest:Players()
            local Target
            local Distance = Huge

            for _, Player in GetPlayers(Players) do
                if LocalPlayer == Player then
                    continue
                end
                --
                local Char = Character:GetCharacters(Player)
                local RootPart = Character:GetRootPart(Player)
                --
                if not Char or not RootPart then
                    continue
                end
                --
                local ScreenPos, OnScreen = Camera:WorldToViewportPoint(RootPart.Position)
                --
                if not OnScreen then
                    continue
                end
                --
                local Magnitude = (Vector2New(ScreenPos.X, ScreenPos.Y) - InputService:GetMouseLocation()).Magnitude
                --
                if Magnitude < Distance then
                    Distance = Magnitude
                    Target = Player
                end
            end
            
            local BotsFolder = FindFirstChild(Workspace, "Bots")
            
            if BotsFolder then
                for _, Bot in GetChildren(BotsFolder) do
                    local RootPart = FindFirstChild(Bot, 'HumanoidRootPart')
                    
                    if not RootPart then
                        continue
                    end
                    --
                    local Humanoid = FindFirstChildOfClass(Bot, 'Humanoid')
                    
                    if not Humanoid or Humanoid.Health <= 0 then
                        continue
                    end
                    --
                    local ScreenPos, OnScreen = Camera:WorldToViewportPoint(RootPart.Position)
                    
                    if not OnScreen then
                        continue
                    end
                    --
                    local Magnitude = (Vector2New(ScreenPos.X, ScreenPos.Y) - InputService:GetMouseLocation()).Magnitude
                    --
                    if Magnitude < Distance then
                        Distance = Magnitude
                        Target = Bot
                    end
                end
            end
            --
            return Target
        end

        function Closest:Parts(Player)
            local Distance = Huge
            local TargetParts
            --
            for _, ClosestPart in Player:GetChildren() do
                if not ClosestPart:IsA("BasePart") then
                    continue
                end
                --
                local ScreenPos, OnScreen = Camera:WorldToViewportPoint(ClosestPart.Position)
                --
                if not OnScreen then
                    continue
                end
                --
                local Magnitude = (Vector2New(ScreenPos.X, ScreenPos.Y) - InputService:GetMouseLocation()).Magnitude
                --
                if Magnitude < Distance then
                    Distance = Magnitude
                    TargetParts = ClosestPart
                end
            end
            
            return TargetParts
        end

        function Closest:Points(Part, Mode, Scale)
            local UnitRay = Camera:ViewportPointToRay(GetMouseLocation(InputService).X, GetMouseLocation(InputService).Y)
            local PartCFrame = Part.CFrame
            local PartHalfSize = Part.Size * 0.5
            --
            if Mode == "ZZZ" then
                local ProjectedLocal = PartCFrame:PointToObjectSpace(
                    UnitRay.Origin + UnitRay.Direction * UnitRay.Direction:Dot(PartCFrame.Position - UnitRay.Origin)
                )
                return PartCFrame:PointToWorldSpace(Vector3New(
                    Max(-PartHalfSize.X, Min(PartHalfSize.X, ProjectedLocal.X)),
                    Max(-PartHalfSize.Y, Min(PartHalfSize.Y, ProjectedLocal.Y)),
                    Max(-PartHalfSize.Z, Min(PartHalfSize.Z, ProjectedLocal.Z))
                ))
            elseif Mode == "Scalar" then
                local ScaledHalfSize = Part.Size * (Scale / 2)
                local ProjectedLocal = PartCFrame:PointToObjectSpace(
                    UnitRay.Origin + UnitRay.Direction * UnitRay.Direction:Dot(PartCFrame.Position - UnitRay.Origin)
                )
                return PartCFrame:PointToWorldSpace(Vector3New(
                    Max(-ScaledHalfSize.X, Min(ScaledHalfSize.X, ProjectedLocal.X)),
                    Max(-ScaledHalfSize.Y, Min(ScaledHalfSize.Y, ProjectedLocal.Y)),
                    Max(-ScaledHalfSize.Z, Min(ScaledHalfSize.Z, ProjectedLocal.Z))
                ))
            elseif Mode == "Raycast" then
                local Params = RaycastParams.new()
                Params.FilterType = Enum.RaycastFilterType.Include
                Params.FilterDescendantsInstances = { Part }
                --
                local Result = Workspace:Raycast(UnitRay.Origin, UnitRay.Direction * 1000, Params)
                
               return (Result and Result.Position) or Part.Position
            end
        end
    end

    local Assist = { }; do
        function Assist:AimAssistPosition(Player)
            local GetCharacter = Character:GetCharacters(Player)
            --
            if not GetCharacter then
                return
            end
            --
            local Config = shared.prada.Combat.Camlock
            local HitPartConfig = Config['Hitpart']
            local PredConfig = Config['Prediction']
            local YAxisConfig = PredConfig['Vertical Axis']
            --
            local TargetPart
            local Position
            --
            local HitType = tostring(HitPartConfig.Type or "Head")
            if HitType == "Closest Point" then
                TargetPart = Closest:Parts(GetCharacter)
                --
                if not TargetPart then
                    return
                end
                --
                local CP = HitPartConfig["Closest Point"]
                local ScaleVal = type(CP.Scale) == "table" and CP.Scale[1] or (CP.Scale or 0.5)
                Position = Closest:Points(TargetPart, "Scalar", ScaleVal) or TargetPart.Position
            elseif HitType == "Closest Part" then
                TargetPart = Closest:Parts(GetCharacter)
                --
                if not TargetPart then
                    return
                end
                --
                Position = TargetPart.Position
            else
                TargetPart = FindFirstChild(GetCharacter, HitType) or FindFirstChild(GetCharacter, "Head")
                --
                if not TargetPart then
                    return
                end
                --
                Position = TargetPart.Position
            end
            --
            local Velocity = TargetPart.AssemblyLinearVelocity
            local VelocityY = Velocity.Y
            --
            if YAxisConfig.Override == "Full" then
                VelocityY = 0
            elseif YAxisConfig.Override == "Partial" then
                VelocityY = Max(-YAxisConfig.Clamp, Min(YAxisConfig.Clamp, VelocityY))
            end
            --
            local PredictedPos = Vector3New(
                Position.X + (Velocity.X * PredConfig.Horizontal),
                Position.Y + (VelocityY * PredConfig.Vertically),
                Position.Z + (Velocity.Z * PredConfig['Depth Axis'])
            )
            --
            return PredictedPos, TargetPart
        end

            -- Sentry-mode (Rage): force a tight head lock regardless of configured hitpart.
            if Script.Status.SilentAim.Sentry then
                HitType = "Head"
            end
            if HitType == "Closest Point" then
                TargetPart = Closest:Parts(GetCharacter)
                --
                if not TargetPart then
                    return
                end
                --
                local CP = HitPartConfig["Closest Point"]
                local ScaleVal = type(CP.Scale) == "table" and CP.Scale[1] or (CP.Scale or 0.5)
                Position = Closest:Points(TargetPart, "Scalar", ScaleVal) or TargetPart.Position
            elseif HitType == "Closest Part" then
                TargetPart = Closest:Parts(GetCharacter)
                --
                if not TargetPart then
                    return
                end
                --
                Position = TargetPart.Position
            else
                TargetPart = FindFirstChild(GetCharacter, HitType) or FindFirstChild(GetCharacter, "Head")
                --
                if not TargetPart then
                    return
                end
                --
                Position = TargetPart.Position
            end
            --
            local Velocity = TargetPart.AssemblyLinearVelocity
            local VelocityY = Velocity.Y
            --
            if YAxisConfig.Override == "Full" then
                VelocityY = 0
            elseif YAxisConfig.Override == "Partial" then
                VelocityY = Max(-YAxisConfig.Clamp, Min(YAxisConfig.Clamp, VelocityY))
            end
            --
            local PredictedPos = Vector3New(
                Position.X + (Velocity.X * PredConfig.Horizontal),
                Position.Y + (VelocityY * PredConfig.Vertically),
                Position.Z + (Velocity.Z * PredConfig['Depth Axis'])
            )
            --
            -- Future: extra whole-body lead ticks on top of base prediction.
            local FutureConfig = Config.Future
            if FutureConfig and FutureConfig.Enabled then
                local FutureTicks = FutureConfig.Settings and FutureConfig.Settings.Value or 0
                if FutureTicks and FutureTicks > 0 then
                    local LeadVel = Vector3New(Velocity.X, VelocityY, Velocity.Z)
                    PredictedPos = PredictedPos + LeadVel * (PredConfig.Horizontal * FutureTicks)
                end
            end
            --
            return PredictedPos, TargetPart
        end

        function Assist:TriggerBot(Target)
            local Config = shared.prada.Combat.Triggerbot
            local State = Script.Status.TriggerBot
            local GetCharacter = Character:GetCharacters(Target)
            --
            if not GetCharacter then
                return
            end
            --
            local RootPart = Character:GetRootPart(Target)
            --
            if not RootPart then
                return
            end
            --
            local PredConfig = Config['Prediction']
            local YAxisConfig = PredConfig['Vertical Axis']
            local Velocity = RootPart.AssemblyLinearVelocity
            local VelocityY = Velocity.Y
            --
            if YAxisConfig.Override == "Full" then
                VelocityY = 0
            elseif YAxisConfig.Override == "Partial" then
                VelocityY = Max(-YAxisConfig.Clamp, Min(YAxisConfig.Clamp, VelocityY))
            end
            --
            local PredictedPos = Vector3New(
                RootPart.Position.X + (Velocity.X * PredConfig.Horizontal),
                RootPart.Position.Y + (VelocityY * PredConfig.Vertically),
                RootPart.Position.Z + (Velocity.Z * PredConfig['Depth Axis'])
            )
            --
            local InRange = false
            --
            if Config.Mode == "Raycast" then
                local UnitRay = Camera:ViewportPointToRay(GetMouseLocation(InputService).X, GetMouseLocation(InputService).Y)
                --
                local Params = RaycastParams.new()
                Params.FilterType = Enum.RaycastFilterType.Include
                Params.FilterDescendantsInstances = { GetCharacter }
                --
                local Result = Workspace:Raycast(UnitRay.Origin, UnitRay.Direction * 1000, Params)
                InRange = Result and Result.Instance ~= nil
            elseif Config.Mode == "Fov" then
                local FovConfig = shared.prada.Combat.Triggerbot["Box-FOV"]
                InRange = Utility:IsHovered(PredictedPos, FovConfig)
            end
            --
            if not InRange then
                return
            end
            --
            local Now = os.clock()
            --
            local LocalCharacter = Character:GetCharacters(LocalPlayer)
            --
            if not LocalCharacter then
                return
            end
            --
            local Tool = FindFirstChildOfClass(LocalCharacter, "Tool")
            --
            if not Tool then
                return
            end

            -- > Whitelist gate: every weapon is always allowed (Deagle + all guns).
            -- local TriggerWhitelist = Config.Whitelisted
            -- if TriggerWhitelist and TriggerWhitelist[Tool.Name] == false then
            --     Script.Status.TriggerBot.ActiveTarget = nil
            --     return
            -- end

            -- > Max-distance-per-gun check
            local RangeCheck = Config["Range-Check"]
            local DistanceConfig = RangeCheck or { Enabled = true, ["Use Gun Range"] = true, Override = 200 }
            local DistanceEnabled = DistanceConfig.Enabled == true and DistanceConfig.Enabled ~= nil or (RangeCheck ~= nil)
            local MaxRange = DistanceConfig.Override or 200
            local Handle = Tool:FindFirstChild("Handle")
            local Origin = Handle and Handle.Position or RootPart.Position
            --
            -- > Per-gun Data override wins over the shared Override when nonzero.
            local PerGunRange = DistanceConfig.Data and DistanceConfig.Data[Tool.Name]
            if PerGunRange and PerGunRange ~= 0 then
                MaxRange = PerGunRange
            end
            --
            if DistanceConfig['Use Gun Range'] then
                local RangeValue = Tool:FindFirstChild("Range")
                if RangeValue and typeof(RangeValue.Value) == "number" then
                    MaxRange = RangeValue.Value
                end
            end
            --
            if (DistanceEnabled or DistanceConfig.Override) and (Origin - PredictedPos).Magnitude > MaxRange then
                Script.Status.TriggerBot.ActiveTarget = nil
                return
            end
            --
            -- > Gate on the gun's real fire cooldown so we don't spam faster than it can shoot
            local GunCooldown = 0.1
            local CooldownValue = Tool:FindFirstChild("ShootingCooldown")
            if CooldownValue and typeof(CooldownValue.Value) == "number" then
                GunCooldown = CooldownValue.Value
            end
            --
            -- > Per-gun click delay (prada Delay is {min, max} in milliseconds)
            local GunDelay = Config.Delay and Config.Delay[Tool.Name]
            local MinCd = 0.02
            local MaxCd = 0.05
            if type(GunDelay) == "table" then
                MinCd = (tonumber(GunDelay[1]) or 20) / 1000
                MaxCd = (tonumber(GunDelay[2]) or 50) / 1000
            end
            local FireInterval = Max(GunCooldown, MinCd + (math.random() * math.max(0, MaxCd - MinCd)))
            --
            if Now < State.NextFireAt then
                return
            end
            --
            Tool:Activate()
            --
            State.NextFireAt = Now + FireInterval
        end
    end

    local Speed = { } ; do
        Speed.BaseSpeed = 16
        Speed.LastSetSpeed = nil
        Speed.LastState = nil
        Speed.IsShooting = false
        Speed.LastAmmo = 0
        Speed.LastTool = nil
        Speed.LastShotTime = 0
        Speed.TargetSpeed = nil      -- the boosted value the re-assert hook maintains
        Speed.BoundHumanoid = nil    -- humanoid the re-assert hook is currently bound to
        Speed.ReassertConn = nil     -- GetPropertyChangedSignal("WalkSpeed") connection

        function Speed:Capture(Humanoid)
            local CurrentState = Humanoid:GetState()

            if CurrentState == Enum.HumanoidStateType.Jumping
                or CurrentState == Enum.HumanoidStateType.Freefall then
                return
            end

            local CurrentSpeed = math.round(Humanoid.WalkSpeed * 1000)
            local LastSpeed = -1
            
            if self.LastSetSpeed then
                LastSpeed = math.round(self.LastSetSpeed * 1000)
            end
            
            if CurrentSpeed ~= LastSpeed and CurrentSpeed > 0 then
                self.BaseSpeed = Humanoid.WalkSpeed
                self.LastSetSpeed = Humanoid.WalkSpeed
            end
        end

        function Speed:Reset(Humanoid)
            -- Stop the re-assert hook from fighting once the toggle is off.
            self.TargetSpeed = nil

            local CurrentState = Humanoid:GetState()

            if CurrentState == Enum.HumanoidStateType.Jumping
                or CurrentState == Enum.HumanoidStateType.Freefall then
                return
            end

            local CurrentSpeed = math.round(Humanoid.WalkSpeed * 1000)
            local LastSpeed = -1
            
            if self.LastSetSpeed then
                LastSpeed = math.round(self.LastSetSpeed * 1000)
            end
            
            if CurrentSpeed == LastSpeed and self.LastSetSpeed ~= self.BaseSpeed then
                Humanoid.WalkSpeed = self.BaseSpeed
                self.LastSetSpeed = self.BaseSpeed
            end
        end

        function Speed:DetectState(PlayerCharacter, Humanoid)
            local Tool = FindFirstChildOfClass(PlayerCharacter, 'Tool')
            local IsKnife = false
            local IsReload = false

            if Tool then
                local Name = Tool.Name
                local LowerName = Name:lower()
                IsKnife = (Name == '[Knife]') or (LowerName:find('knife') ~= nil)

                local Ammo = FindFirstChild(Tool, 'Ammo')

                if Ammo then
                    if self.LastTool == Tool and Ammo.Value < self.LastAmmo then
                        self.LastShotTime = os.clock()
                    end
                    
                    self.LastAmmo = Ammo.Value

                    if Ammo.Value == 0 then
                        IsReload = true
                    end
                end
                
                self.LastTool = Tool

                if not IsReload then
                    local Reloading = FindFirstChild(Tool, 'Reloading')
                    
                    if Reloading and Reloading.Value then
                        IsReload = true
                    end
                end
            else
                self.LastTool = nil
            end

            local BodyEffects = FindFirstChild(PlayerCharacter, 'BodyEffects')

            if BodyEffects then
                local Reload = FindFirstChild(BodyEffects, 'Reload')
                
                if Reload and Reload.Value == true then
                    IsReload = true
                end
            end

            local IsLowHealth = false
            
            if Humanoid and Humanoid.Health > 0 and Humanoid.Health <= 25 then
                IsLowHealth = true
            end

            if (os.clock() - self.LastShotTime) <= 0.5 then
                self.IsShooting = true
            else
                self.IsShooting = false
            end

            if IsKnife then
                return 'Knife'
            elseif IsLowHealth then
                return 'Low Health'
            elseif IsReload then
                return 'Reloading'
            elseif self.IsShooting then
                return 'Shooting'
            end

            return 'Default'
        end

        function Speed:Calculate(Method, Values)
            if Method == 'Multiplication' then
                return self.BaseSpeed * Values
            elseif Method == 'Override' then
                return Values
            elseif Method == 'Addition' then
                return self.BaseSpeed + Values
            elseif Method == 'Subtraction' then
                return math.max(0, self.BaseSpeed - Values)
            elseif Method == 'Percentage' then
                return self.BaseSpeed * (Values / 100)
            elseif Method == 'Division' then
                if Values ~= 0 then
                    return self.BaseSpeed / Values
                end
                
                return self.BaseSpeed
            elseif Method == 'Power' then
                return self.BaseSpeed ^ Values
            end

            return self.BaseSpeed
        end

        function Speed:Apply()
            local PlayerCharacter = Character:GetCharacters(LocalPlayer)
            local Humanoid = nil

            if PlayerCharacter then
                Humanoid = Character:GetHumanoid(LocalPlayer)
            end

            if not PlayerCharacter or not Humanoid then
                return
            end

            -- Bind the WalkSpeed re-assert hook. Da Hood's server resets
            -- Humanoid.WalkSpeed back to 16 constantly, so writing it once per
            -- frame (RenderStepped) always loses the race. Instead we bind a
            -- GetPropertyChangedSignal("WalkSpeed") handler that instantly gives
            -- the boosted value back the moment the game resets it — this beats
            -- the reset and makes the speed actually hold. Re-binds on respawn.
            if self.BoundHumanoid ~= Humanoid then
                if self.ReassertConn then
                    self.ReassertConn:Disconnect()
                    self.ReassertConn = nil
                end
                self.BoundHumanoid = Humanoid
                self.ReassertConn = Humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
                    if self.TargetSpeed and Script.Status.WalkSpeed.Toggles
                        and Humanoid.WalkSpeed ~= self.TargetSpeed then
                        Humanoid.WalkSpeed = self.TargetSpeed
                    end
                end)
            end

            if not self.LastSetSpeed then
                self.LastSetSpeed = Humanoid.WalkSpeed
                self.BaseSpeed = Humanoid.WalkSpeed
            end

            self:Capture(Humanoid)

            local SpeedMain = shared.prada.Character.Modifications.Speed

            if not SpeedMain.Enabled or not Script.Status.WalkSpeed.Toggles then
                self:Reset(Humanoid)
                return
            end

            local State = self:DetectState(PlayerCharacter, Humanoid)
            local Data = SpeedMain.Data and SpeedMain.Data.Speed or {}
            -- Apply the FULL Normal speed in every state so WalkSpeed
            -- works at max speed whether shooting, low health, reloading,
            -- holding a knife, or just running normally.
            local StateKey = {
                Default = "Normal",
                ["Low Health"] = "Normal",
                Reloading = "Normal",
                Shooting = "Normal",
                Knife = "Normal",
            }
            local Value = Data[StateKey[State]]
            if Value == nil then
                Value = Data.Normal or 1
            end

            local NewSpeed = self:Calculate("Addition", Value)

            -- Hard engine cap. Da Hood's third-person camera can only track the
            -- character up to ~60 WalkSpeed; past that the view physically
            -- freezes no matter what the script does. 60 = 4x the 16 default,
            -- clearly fast, but still inside the window the camera follows
            -- smoothly. Config values like 520 can't be reached without the
            -- camera breaking -- that's a game-engine limit, not a code bug.
            if NewSpeed > 60 then
                NewSpeed = 60
            end

            self.TargetSpeed = NewSpeed
            if math.round(NewSpeed * 1000) ~= math.round(Humanoid.WalkSpeed * 1000) then
                Humanoid.WalkSpeed = NewSpeed
                self.LastSetSpeed = NewSpeed
            end
        end

        function Speed:AntiTrip()
            local SpeedMain = shared.prada.Character.Modifications.Speed
            if not SpeedMain.Enabled or SpeedMain["Anti Trip"] ~= true then
                return
            end

            local PlayerCharacter = Character:GetCharacters(LocalPlayer)
            local Humanoid = nil

            if PlayerCharacter then
                Humanoid = Character:GetHumanoid(LocalPlayer)
            end

            if not PlayerCharacter or not Humanoid then
                return
            end

            local CurrentState = Humanoid:GetState()

            if CurrentState == Enum.HumanoidStateType.FallingDown
                or CurrentState == Enum.HumanoidStateType.Ragdoll then
                if self.LastState ~= Enum.HumanoidStateType.Jumping
                    and self.LastState ~= Enum.HumanoidStateType.Freefall
                    and self.LastState ~= Enum.HumanoidStateType.Landed then
                    Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                end
            end

            self.LastState = CurrentState
        end
    end

    local Panic = { }; do
        -- Smoothly drops the local character straight down to solid ground, no
        -- matter the height. Slides the full distance (never stops short), keeps
        -- the character's exact facing (never spins sideways), and settles softly
        -- via an ease-out so the landing is silky instead of a clunk.
        function Panic:Ground()
            local PlayerCharacter = Character:GetCharacters(LocalPlayer)
            local RootPart = Character:GetRootPart(LocalPlayer)
            local Humanoid = Character:GetHumanoid(LocalPlayer)

            if not PlayerCharacter or not RootPart or not Humanoid then
                return false
            end

            local Params = RaycastParams.new()
            Params.FilterType = Enum.RaycastFilterType.Exclude
            Params.FilterDescendantsInstances = { PlayerCharacter }
            Params.IgnoreWater = true

            local Origin = RootPart.Position
            local Down = Vector3New(0, -1, 0)
            local MaxFall = 100000

            local Result = Workspace:Raycast(Origin, Down * MaxFall, Params)
            if not (Result and Result.Position) then
                return false
            end

            -- Feet land on the grounded surface with a hair of clearance: high
            -- enough not to clip into it, low enough not to hover over it.
            local TargetY = Result.Position.Y + (RootPart.Size.Y / 2) + 0.5

            -- INSTANT drop: snap straight down to the ground in a single frame.
            -- No easing, no glide — one press and you're already on the floor.
            if RootPart:IsA("BasePart") then
                RootPart.AssemblyLinearVelocity = Vector3New(0, 0, 0)
                RootPart.AssemblyAngularVelocity = Vector3New(0, 0, 0)
            end

            -- Keep the exact facing; only the vertical position changes instantly.
            local Rot = RootPart.CFrame
            RootPart.CFrame = Rot * CFrameNew(0, TargetY - RootPart.Position.Y, 0)

            pcall(function()
                local St = Humanoid:GetState()
                if St == Enum.HumanoidStateType.FallingDown
                    or St == Enum.HumanoidStateType.Freefall
                    or St == Enum.HumanoidStateType.Ragdoll
                    or St == Enum.HumanoidStateType.Jumping then
                    Humanoid:ChangeState(Enum.HumanoidStateType.Running)
                end
            end)

            return true
        end
    end

        if RageConfig and RageConfig.Enabled then
            local RageMult = RageConfig.Settings and RageConfig.Settings.Multiplier or 0
            local ScaleF = 1 + (RageMult or 0)
            SentryActive = RageConfig.Settings and RageConfig.Settings["Sentry-Mode"] == true
            LockFov = {
                Width = type(Fov.Width) == "table" and { Fov.Width[1] * ScaleF, Fov.Width[2] * ScaleF } or Fov.Width * ScaleF,
                Height = type(Fov.Height) == "table" and { Fov.Height[1] * ScaleF, Fov.Height[2] * ScaleF } or Fov.Height * ScaleF,
            }
        end
        Script.Status.SilentAim.Sentry = SentryActive
        
        local SelectedTarget
        
        if SelectionSystem == "Target" then
            SelectedTarget = Script.Status.SilentAim.Target
        else
            SelectedTarget = Closest:Players()
        end
        
        if not SelectedTarget then
            Script.Status.SilentAim.Target = nil
            Script.Status.SilentAim.ActiveTarget = nil
            FOV:Update(SilentAimBox, nil, Fov)
            return
        end
        
        local CharacterModel = Character:GetCharacters(SelectedTarget)
        
        if not CharacterModel then
            Script.Status.SilentAim.Target = nil
            Script.Status.SilentAim.ActiveTarget = nil
            
            if SelectionSystem == "Target" then
                Script.Status.Target = false 
            end
            
            FOV:Update(SilentAimBox, nil, Fov)
            return
        end
        
        if Checks.Knocked then
            local TargetHumanoid = Character:GetHumanoid(SelectedTarget)
            local IsKnockedOut = Games:GetKoValues(SelectedTarget)
            
            if (TargetHumanoid and TargetHumanoid.Health <= 1) or IsKnockedOut then
                Script.Status.SilentAim.Target = nil
                Script.Status.SilentAim.ActiveTarget = nil
                
                if SelectionSystem == "Target" then
                    Script.Status.Target = false
                end
                
                FOV:Update(SilentAimBox, nil, Fov)
                return
            end
        end
        
        if Checks.Grabbed then
            if Games:GetGrabbedValues(SelectedTarget) then
                Script.Status.SilentAim.Target = nil
                Script.Status.SilentAim.ActiveTarget = nil
                
                if SelectionSystem == "Target" then
                    Script.Status.Target = false 
                end
                
                FOV:Update(SilentAimBox, nil, Fov)
                return
            end
        end
        
        local CanAim = true
        if VisibleCheck then
            if not Character:GetVisible(SelectedTarget) then
                CanAim = false
                if not ShowFovCheck then
                    Script.Status.SilentAim.ActiveTarget = nil
                    FOV:Update(SilentAimBox, nil, Fov)
                    return
                end
            end
        end
        
        local RootPart = Character:GetRootPart(SelectedTarget)
        
        if not RootPart then
            Script.Status.SilentAim.Target = nil
            Script.Status.SilentAim.ActiveTarget = nil
            
            if SelectionSystem == "Target" then
                Script.Status.Target = false 
            end
            
            FOV:Update(SilentAimBox, nil, Fov)
            return
        end
        
        Script.Status.SilentAim.Target = SelectedTarget
        FOV:Update(SilentAimBox, RootPart.Position, Fov)
        
        if not Silent.Enabled or not Script.Status.SilentAim.Toggles or not CanAim then
            Script.Status.SilentAim.ActiveTarget = nil
            return
        end
        
        if Utility:IsHovered(RootPart.Position, LockFov) then
            Script.Status.SilentAim.ActiveTarget = SelectedTarget
        else
            Script.Status.SilentAim.ActiveTarget = nil
        end
    end

    function Assist:HandleAimAssist()
        local AimAssist = shared.prada.Combat.Camlock
        local SelectionSystem = shared.prada.Main["Selection System"]
        local Checks = shared.prada.Main.Checks
        local Fov = shared.prada.Combat.Camlock["Box-FOV"]
        local VisibleCheck = false
        local ShowFovCheck = false
        
        local SelectedTarget
        
        if SelectionSystem == "Target" then
            SelectedTarget = Script.Status.AimAssist.Target
        else
            SelectedTarget = Closest:Players()
        end
        
        if not SelectedTarget then
            Script.Status.AimAssist.Target = nil
            Script.Status.AimAssist.ActiveTarget = nil
            FOV:Update(AimAssistBox, nil, Fov)
            return
        end
        
        local CharacterModel = Character:GetCharacters(SelectedTarget)
        
        if not CharacterModel then
            Script.Status.AimAssist.Target = nil
            Script.Status.AimAssist.ActiveTarget = nil
            
            if SelectionSystem == "Target" then
                Script.Status.Target = false 
            end
            
            FOV:Update(AimAssistBox, nil, Fov)
            return
        end
        
        if Checks.Knocked then
            local TargetHumanoid = Character:GetHumanoid(SelectedTarget)
            local IsKnockedOut = Games:GetKoValues(SelectedTarget)
            
            if (TargetHumanoid and TargetHumanoid.Health <= 1) or IsKnockedOut then
                Script.Status.AimAssist.Target = nil
                Script.Status.AimAssist.ActiveTarget = nil
                
                if SelectionSystem == "Target" then
                    Script.Status.Target = false 
                end
                
                FOV:Update(AimAssistBox, nil, Fov)
                return
            end
        end
        
        if Checks.Grabbed then
            if Games:GetGrabbedValues(SelectedTarget) then
                Script.Status.AimAssist.Target = nil
                Script.Status.AimAssist.ActiveTarget = nil
                
                if SelectionSystem == "Target" then
                    Script.Status.Target = false 
                end
                
                FOV:Update(AimAssistBox, nil, Fov)
                return
            end
        end
        
        local CanAim = true

        if VisibleCheck then
            if not Character:GetVisible(SelectedTarget) then
                CanAim = false

                if not ShowFovCheck then
                    Script.Status.AimAssist.ActiveTarget = nil
                    FOV:Update(AimAssistBox, nil, Fov)
                    return
                end
            end
        end
        
        local RootPart = Character:GetRootPart(SelectedTarget)
        
        if not RootPart then
            Script.Status.AimAssist.Target = nil
            Script.Status.AimAssist.ActiveTarget = nil
            
            if SelectionSystem == "Target" then
                Script.Status.Target = false 
            end
            
            FOV:Update(AimAssistBox, nil, Fov)
            return
        end

        -- > Camlock Whitelist + Range-Check: gate the lock on the currently equipped gun.
        local CamLocked = true
        local LocalChar = Character:GetCharacters(LocalPlayer)
        local EquippedTool = LocalChar and FindFirstChildOfClass(LocalChar, "Tool")
        if EquippedTool then
            -- > Camlock whitelist: every weapon is always allowed (Deagle + all guns).
            -- local CamWhitelist = AimAssist.Whitelisted
            -- if CamWhitelist and CamWhitelist[EquippedTool.Name] == false then
            --     CamLocked = false
            -- end

            if CamLocked then
                local CamRangeCheck = AimAssist["Range-Check"]
                if CamRangeCheck and CamRangeCheck.Enabled then
                    local CamMax = CamRangeCheck.Override or 200
                    local CamPerGun = CamRangeCheck.Data and CamRangeCheck.Data[EquippedTool.Name]
                    if CamPerGun and CamPerGun ~= 0 then
                        CamMax = CamPerGun
                    end
                    local CamHandle = EquippedTool:FindFirstChild("Handle")
                    local CamOrigin = CamHandle and CamHandle.Position or (LocalChar and LocalChar.PrimaryPart and LocalChar.PrimaryPart.Position)
                    if CamOrigin and (CamOrigin - RootPart.Position).Magnitude > CamMax then
                        CamLocked = false
                    end
                end
            end
        end
        
        Script.Status.AimAssist.Target = SelectedTarget
        FOV:Update(AimAssistBox, RootPart.Position, Fov)
        
        if not AimAssist.Enabled or not Script.Status.AimAssist.Toggles or not CanAim or not CamLocked then
            Script.Status.AimAssist.ActiveTarget = nil
            return
        end
        
        if Utility:IsHovered(RootPart.Position, Fov) then
            Script.Status.AimAssist.ActiveTarget = SelectedTarget
            local PredictedPosition = self:AimAssistPosition(SelectedTarget)
            
            if PredictedPosition then
                -- Anchor the aim camera at the LOCAL character so it follows
                -- your movement instead of freezing in place after you walk.
                -- The old code pinned the origin to Camera.CFrame.Position,
                -- which never updates, so with CameraType=Custom the camera
                -- stayed locked at its last position. Re-anchoring it on the
                -- moving root each frame keeps the look on target while the
                -- camera tracks you.
                local LocalRoot = Character:GetRootPart(LocalPlayer)
                local Origin = LocalRoot and LocalRoot.Position or Camera.CFrame.Position
                local Look = PredictedPosition - Origin
                if Look.Magnitude > 0.01 then
                    Look = Look.Unit
                end
                -- sit a bit behind/above the character, facing the target
                local TargetCFrame = CFrameNew(Origin + Look * -3 + Vector3New(0, 1.5, 0), PredictedPosition)
                local Alpha = AimAssist.Smoothness
                local Curves = AimAssist["Smoothness Curves"]

                if Curves then
                    local Style = Enum.EasingStyle[Curves.Style] or Enum.EasingStyle.Linear
                    local Direction = Enum.EasingDirection[Curves.Direction] or Enum.EasingDirection.InOut
                    Alpha = TweenService:GetValue(Alpha, Style, Direction)
                end
                
                Camera.CFrame = Camera.CFrame:Lerp(TargetCFrame, Alpha)
            end
        else
            Script.Status.AimAssist.ActiveTarget = nil
        end
    end

        function ExtraSensoryPerception:Update()
            local RightNow = os.clock()

            if RightNow - LastUpdate < FrameRate then 
                return 
            end

            LastUpdate = RightNow

            if not shared.prada.Visuals.ESP.Enabled or not Script.Status.Esp.Toggles then
                for _, Label in Cache do
                    for _, Property in Label do
                        Property.Visible = false
                    end
                end
                
                return

            if not EspConfig.Enabled then
                for _, Label in Cache do

    -- Ghost-feature handlers: Character.Modifications (Headless/Ramp/Auto-Reset/Spiderman/Anti-Future)
    -- These keys exist in the prada table but previously had no backing code.
    local CharacterGhosts = { }; do
        CharacterGhosts.BaseHeadTransparency = {}
        CharacterGhosts.RampSpeed = nil
        CharacterGhosts.LastHorizontalSpeed = 0

        local function GetLocalHumanoid(Player)
            -- Return ONLY the Humanoid. (This helper used to return (Char, Humanoid); callers
            -- captured just the first value and got the Model instead of the Humanoid, which made
            -- Humanoid:GetState() / WalkSpeed() throw and killed the whole RenderStepped callback.)
            return Character:GetHumanoid(Player)
        end

        function CharacterGhosts:ApplyHeadless(Player)
            local Char = Character:GetCharacters(Player)
            local Head = Char and FindFirstChild(Char, "Head")
            local HeadlessCfg = shared.prada.Character.Modifications.Headless
            if not Head then return end

            if self.BaseHeadTransparency[Head] == nil then
                self.BaseHeadTransparency[Head] = Head.Transparency
            end

            local Enabled = HeadlessCfg == true
            Head.Transparency = (Enabled and 1 or self.BaseHeadTransparency[Head])
            if Enabled then
                local Mesh = FindFirstChild(Head, "Mesh") or FindFirstChildOfClass(Head, "SpecialMesh")
                if Mesh then Mesh.Transparency = 1 end
            end
        end

        function CharacterGhosts:ApplyRamp(Player)
            local SpeedMain = shared.prada.Character.Modifications.Speed
            local Ramp = SpeedMain.Ramp
            local Humanoid = GetLocalHumanoid(Player)
            local Char = Character:GetCharacters(Player)

            if not Humanoid or not Char or not Ramp then return end
            if not Ramp.Enabled or not SpeedMain.Enabled or not Script.Status.WalkSpeed.Toggles then
                self.LastHorizontalSpeed = 0
                return
            end

            local RootPart = Character:GetRootPart(Player)
            if not RootPart then return end

            local Horizontal = Vector3New(RootPart.AssemblyLinearVelocity.X, 0, RootPart.AssemblyLinearVelocity.Z).Magnitude
            local Amount = Ramp.Amount or 0

            -- Ramp up while running, decay to base when not.
            if Horizontal > 0.5 and Humanoid:GetState() == Enum.HumanoidStateType.Running then
                self.RampSpeed = self.RampSpeed and (self.RampSpeed + Amount) or (Humanoid.WalkSpeed + Amount)
                Humanoid.WalkSpeed = math.min(self.RampSpeed, Humanoid.WalkSpeed + Amount)
            else
                self.RampSpeed = nil
                if self.LastHorizontalSpeed == 0 then
                    Humanoid.WalkSpeed = math.max(Humanoid.WalkSpeed - Amount, Humanoid.WalkSpeed)
                end
            end
            self.LastHorizontalSpeed = Horizontal
        end

        function CharacterGhosts:ApplyAutoReset(Player)
            local AutoReset = shared.prada.Character.Modifications["Auto-Reset"]
            local Humanoid = GetLocalHumanoid(Player)
            if not AutoReset or not AutoReset.Enabled or not Humanoid then return end

            local Settings = AutoReset.Settings or {}
            local ShouldReset = false
            local When = Settings["When"] or "Knocked"

            if When == "Knocked" then
                local IsKnocked = Games:GetKoValues(Player)
                if IsKnocked then
                    ShouldReset = true
                end
            elseif When == "MissingHealth" then
                local MinHealth = Settings["Min-Health"] or 10
                if Humanoid.Health <= MinHealth then
                    ShouldReset = true
                end
            end

            if ShouldReset and Humanoid.Health > 0 then
                Humanoid.Health = 0
            end
        end

        function CharacterGhosts:ApplyAntiFuture(Player)
            local AntiFuture = shared.prada.Character.Modifications["Anti-Future"]
            if not AntiFuture or not AntiFuture.Enabled then return end

            local Settings = AntiFuture.Settings or {}
            local Mode = Settings["Mode"] or "Velocity"
            local Multiplier = Settings["Multiplier"] or Vector3New(1.25, -5, 1.25)

            local Char = Character:GetCharacters(Player)
            local Humanoid = GetLocalHumanoid(Player)
            local RootPart = Character:GetRootPart(Player)
            if not Char or not Humanoid or not RootPart then return end

            -- Only engage during a jump/airborne so we don't fight grounded movement.
            local State = Humanoid:GetState()
            if State ~= Enum.HumanoidStateType.Jumping and State ~= Enum.HumanoidStateType.Freefall then
                return
            end

            -- "Velocity": amplify horizontal lead while resisting upward pop.
            if Mode == "Velocity" then
                local Vel = RootPart.AssemblyLinearVelocity
                RootPart.AssemblyLinearVelocity = Vector3New(
                    Vel.X * Multiplier.X,
                    Vel.Y,
                    Vel.Z * Multiplier.Z
                )
            else
                -- "Legit": a gentler drag so movement reads naturally.
                local Vel = RootPart.AssemblyLinearVelocity
                RootPart.AssemblyLinearVelocity = Vector3New(
                    Vel.X * 0.95,
                    math.min(Vel.Y, 0),
                    Vel.Z * 0.95
                )
            end
        end

        -- Ghost feature: Character.Modifications.Spiderman — wall climb.
        -- Only triggers when you are actually TOUCHING a vertical wall (never the
        -- ground or a ceiling). Once touching, it climbs automatically. Can be
        -- toggled on/off with the Spiderman keybind (Keybinds.Character.Spiderman).
        function CharacterGhosts:ApplySpiderman(Player)
            if Player ~= LocalPlayer then return end
            if not Script.Status.Spiderman.Toggles then return end
            local Spider = shared.prada.Character.Modifications.Spiderman
            if not Spider or not Spider.Enabled then return end

            local Char = Character:GetCharacters(Player)
            local Humanoid = Character:GetHumanoid(Player)
            local RootPart = Character:GetRootPart(Player)
            if not Char or not Humanoid or not RootPart then return end

            local Settings = Spider.Settings or {}
            local Data = Settings.Data or {}
            local Power = Settings.Power or 5

            -- Don't fight flat-ground movement or ragdolls.
            local State = Humanoid:GetState()
            if State == Enum.HumanoidStateType.Ragdoll or State == Enum.HumanoidStateType.Seated then
                return
            end

            -- Direction the chest is facing, flattened to horizontal.
            local Look = RootPart.CFrame.LookVector
            local Forward = Vector3New(Look.X, 0, Look.Z)
            if Forward.Magnitude < 0.001 then Forward = Vector3New(1, 0, 0) end
            Forward = Forward.Unit

            local Params = RaycastParams.new()
            Params.FilterType = Enum.RaycastFilterType.Exclude
            Params.FilterDescendantsInstances = { Char }
            Params.IgnoreWater = true

            -- Cast from torso height a SHORT distance so it only counts as a wall
            -- when your body is genuinely against it — not merely near it.
            local Origin = RootPart.Position + Vector3New(0, -1, 0)
            local Hit = Workspace:Raycast(Origin, Forward * 2, Params)

            if not Hit then
                return -- nothing touching; normal movement
            end

            -- Wall check: a real wall's surface normal is mostly HORIZONTAL
            -- (|normal.Y| small). Flat ground, floors and ceilings have a mostly
            -- VERTICAL normal and are rejected here.
            local IsWall = math.abs(Hit.Normal.Y) < 0.5

            -- And the contact must be around chest height, not your feet brushing
            -- a slope or stair lip.
            local ContactY = Hit.Position.Y
            local TorsoY = RootPart.Position.Y
            if not IsWall or math.abs(ContactY - TorsoY) > 3 then
                return
            end

            -- Touching a real wall. Freefall stops joystick gravity from tearing us
            -- off the wall; then we drive the climb ourselves every frame.
            if State ~= Enum.HumanoidStateType.Freefall then
                Humanoid:ChangeState(Enum.HumanoidStateType.Freefall)
            end

            local Vel = RootPart.AssemblyLinearVelocity
            local ClimbSpeed = 18 + Power * 2
            -- Steady automatic ascent while clinging to the wall (horizontal drift
            -- damped so you stay pressed against it instead of sliding off).
            RootPart.AssemblyLinearVelocity = Vector3New(Vel.X * 0.2, ClimbSpeed, Vel.Z * 0.2)
        end

        function CharacterGhosts:Apply(Player)
            if not Player then return end
            if not shared.prada or not shared.prada.Character or not shared.prada.Character.Modifications then return end
            -- prada "Modifications.Headless" is a plain boolean.
            self:ApplyHeadless(Player)
            self:ApplyRamp(Player)
            self:ApplyAutoReset(Player)
            self:ApplyAntiFuture(Player)
            self:ApplySpiderman(Player)
        end
    end

    -- Ghost feature: Character.Inventory-Sorter — reorder the backpack hotbar to the configured Order.
    local InventorySorter = { }; do
        InventorySorter.AppliedOrder = nil

        function InventorySorter:Sort(Player)
            local SorterCfg = shared.prada.Character["Inventory-Sorter"]
            if not SorterCfg or not SorterCfg.Enabled then
                self.AppliedOrder = nil
                return
            end

            local Order = SorterCfg.Settings and SorterCfg.Settings.Order or {}
            local Backpack = Player and FindFirstChild(Player, "Backpack")
            if not Backpack then return end

            -- Collect current tools keyed by name.
            local Tools = {}
            for _, Child in ipairs(GetChildren(Backpack)) do
                if Child:IsA("Tool") then
                    Tools[Child.Name] = Child
                end
            end

            -- Re-parent in the configured order.
            for Index = 1, #Order do
                local Tool = Tools[Order[Index]]
                if Tool and Tool.Parent ~= Backpack then
                    Tool.Parent = Backpack
                end
            end

            -- Append any tools that weren't listed, in their existing order.
            for _, Child in ipairs(GetChildren(Backpack)) do
                if Child:IsA("Tool") then
                    local Listed = false
                    for Index = 1, #Order do
                        if Order[Index] == Child.Name then Listed = true break end
                    end
                    if not Listed then Child.Parent = Backpack end
                end
            end
        end
    end

        function SkinChanger:ResolveTool(ToolName)
            local PlayerCharacter = Character:GetCharacters(LocalPlayer)
            local Backpack = FindFirstChild(LocalPlayer, "Backpack")
            local WorkspacePlayers = FindFirstChild(Workspace, "Players")
            local WorkspaceFolder = WorkspacePlayers and FindFirstChild(WorkspacePlayers, LocalPlayer.Name)

            local Containers = {PlayerCharacter, Backpack, WorkspaceFolder}

            for Index = 1, #Containers do
                if Containers[Index] then
                    local Tool = FindFirstChild(Containers[Index], ToolName)

                    if Tool and Tool:IsA("Tool") then
                        return Tool
                    end
                end
            end
        end

        function SkinChanger:RestoreFirearmAudio(Handle)
            local ShootSound = FindFirstChild(Handle, "ShootSound")

            if not ShootSound then
                return
            end

            if self.BaseSounds[ShootSound] then
                ShootSound.SoundId = self.BaseSounds[ShootSound]
            end

            if self.BaseAttributes[ShootSound] then
                local Stored = self.BaseAttributes[ShootSound]
                ShootSound:SetAttribute("SequenceSFX", Stored.SequenceSFX)
                ShootSound:SetAttribute("CurrentSequence", Stored.CurrentSequence)
                ShootSound:SetAttribute("DefaultSoundId", Stored.DefaultSoundId)
            end
        end

        function SkinChanger:UpdateFirearmAudio(Handle, ToolName, SkinName)
            local ShootSound = FindFirstChild(Handle, "ShootSound")

            if not ShootSound then
                return
            end

            if not self.BaseSounds[ShootSound] then
                self.BaseSounds[ShootSound] = ShootSound.SoundId
                self.BaseAttributes[ShootSound] = {
                    SequenceSFX = ShootSound:GetAttribute("SequenceSFX"),
                    CurrentSequence = ShootSound:GetAttribute("CurrentSequence"),
                    DefaultSoundId = ShootSound:GetAttribute("DefaultSoundId"),
                }
            end

            local WeaponFolder = FindFirstChild(FirearmSounds, ToolName)

            if not WeaponFolder then
                self:RestoreFirearmAudio(Handle)
                return
            end

            local SoundEntry = FindFirstChild(WeaponFolder, SkinName)

            if not SoundEntry or SoundEntry.Value == "" then
                self:RestoreFirearmAudio(Handle)
                return
            end

            ShootSound.SoundId = SoundEntry.Value

            for Key, Value in SoundEntry:GetAttributes() do
                ShootSound:SetAttribute(Key, Value)
            end
        end

        function SkinChanger:ApplyFirearm(ToolName, SkinName)
            local Tool = self:ResolveTool(ToolName)

            if not Tool then
                return
            end

            local Registry = FirearmRegistry[ToolName]

            if not Registry or not Registry[SkinName] then
                return
            end

            local DefaultPart = FindFirstChild(Tool, "Default")

            if not DefaultPart then
                return
            end

            local Handle = FindFirstChild(Tool, "Handle")
            local ExistingMesh = FindFirstChild(DefaultPart, "Mesh")

            if ExistingMesh then
                ExistingMesh:Destroy()
            end

            if not self.BaseTextures[DefaultPart] then
                self.BaseTextures[DefaultPart] = DefaultPart.TextureID
            end

            local SkinInfo = Registry[SkinName]
            local TextureSource = SkinInfo.TextureID

            if typeof(TextureSource) == "string" then
                DefaultPart.TextureID = TextureSource
                DefaultPart.Transparency = 0
                DefaultPart.LocalTransparencyModifier = 0

            elseif typeof(TextureSource) == "Instance" then
                local ClonedMesh = TextureSource:Clone()
                ClonedMesh.Name = "Mesh"

                local PartToWeld = ClonedMesh:IsA("BasePart") and ClonedMesh or ClonedMesh:FindFirstChildWhichIsA("BasePart", true)

                if PartToWeld then
                    Utility:CreateInstances("Weld", {
                        Part0 = DefaultPart,
                        Part1 = PartToWeld,
                        C0 = SkinInfo.CFrame or CFrameNew(),
                        Parent = PartToWeld
                    })

                    DefaultPart.Transparency = 1
                    DefaultPart.LocalTransparencyModifier = 1
                end

                ClonedMesh.Parent = DefaultPart

                for _, Descendant in ClonedMesh:GetDescendants() do
                    if Descendant:IsA("ParticleEmitter") or Descendant:IsA("Beam") or Descendant:IsA("Trail") then
                        Descendant.Enabled = true
                    end
                end
            end

            if Handle then
                Handle:SetAttribute("SkinName", SkinName)
                self:UpdateFirearmAudio(Handle, ToolName, SkinName)
                
                if SkinAssets then
                    local ParticleFolder = FindFirstChild(SkinAssets, "GunHandleParticle")
                    local SkinParticle = ParticleFolder and FindFirstChild(ParticleFolder, SkinName)
                    local Emitter = SkinParticle and FindFirstChildOfClass(SkinParticle, "ParticleEmitter")
                    
                    if Emitter then
                        for _, Child in GetChildren(Handle) do
                            if Child:IsA("ParticleEmitter") then
                                Child:Destroy()
                            end
                        end
                        
                        local ClonedEmitter = Emitter:Clone()
                        ClonedEmitter.Enabled = true
                        ClonedEmitter.Parent = Handle
                    end
                end
            end
        end

        function SkinChanger:RevertFirearm(ToolName)
            local Tool = self:ResolveTool(ToolName)

            if not Tool then
                return
            end

            local DefaultPart = FindFirstChild(Tool, "Default")

            if not DefaultPart then
                return
            end

            local Mesh = FindFirstChild(DefaultPart, "Mesh")

            if Mesh then
                Mesh:Destroy()
            end

            if self.BaseTextures[DefaultPart] then
                DefaultPart.TextureID = self.BaseTextures[DefaultPart]
            end

            DefaultPart.Transparency = 0
            DefaultPart.LocalTransparencyModifier = 0

            local Handle = FindFirstChild(Tool, "Handle")

            if Handle then
                Handle:SetAttribute("SkinName", nil)
                self:RestoreFirearmAudio(Handle)
            end
        end

        function SkinChanger:RevertMelee(Tool)
            local State = self.MeleeState[Tool]

            if State then
                if State.AnimationTrack then
                    pcall(function()
                        State.AnimationTrack:Stop()
                    end)
                end

                for Index = 1, #State.Welds do
                    if State.Welds[Index] then
                        State.Welds[Index]:Destroy()
                    end
                end

                for Index = 1, #State.Sounds do
                    if State.Sounds[Index] and State.Sounds[Index].Parent then
                        State.Sounds[Index]:Destroy()
                    end
                end
            end

            local DefaultPart = FindFirstChild(Tool, "Default")

            if DefaultPart then
                for _, Child in GetChildren(DefaultPart) do
                    if Child.Name == "Handle.R" or Child:IsA("Model") or (Child:IsA("BasePart") and Child.Name ~= "Default") then
                        Child:Destroy()
                    end
                end

                DefaultPart.Transparency = 0
            end

            self.MeleeState[Tool] = nil
            Tool:SetAttribute("SkinName", nil)
        end

        function SkinChanger:ApplyMelee(PlayerCharacter, Tool, SkinName)
            local MeleeConfig = self.MeleeRegistry[SkinName]

            if not MeleeConfig then
                return
            end

            local Humanoid = FindFirstChildOfClass(PlayerCharacter, "Humanoid")
            local RightHand = FindFirstChild(PlayerCharacter, "RightHand")

            if not Humanoid or not RightHand then
                return
            end

            if Tool:GetAttribute("SkinName") == SkinName and self.MeleeState[Tool] then
                return
            end

            self:RevertMelee(Tool)

            self.MeleeState[Tool] = {
                AnimationTrack = nil,
                Welds = {},
                Sounds = {}
            }

            local State = self.MeleeState[Tool]
            local DefaultPart = FindFirstChild(Tool, "Default")

            if not DefaultPart then
                return
            end

            DefaultPart.Transparency = 1

            local KnifeModel = FindFirstChild(MeleeMeshes, SkinName)

            if not KnifeModel then
                return
            end

            local ClonedKnife = KnifeModel:Clone()
            ClonedKnife.Name = SkinName

            local HandleR = Utility:CreateInstances("Part", {
                Name = "Handle.R",
                Transparency = 1,
                CanCollide = false,
                Anchored = false,
                Size = Vector3New(0.001, 0.001, 0.001),
                Massless = true,
                Parent = DefaultPart
            })

            Utility:CreateInstances("Motor6D", {
                Name = "Handle.R",
                Part0 = RightHand,
                Part1 = HandleR,
                Parent = HandleR
            })

            local Offset = CFrameNew(MeleeConfig.PositionOffset) * CFrame.Angles(
                Rad(MeleeConfig.RotationOffset.X),
                Rad(MeleeConfig.RotationOffset.Y),
                Rad(MeleeConfig.RotationOffset.Z)
            )

            if ClonedKnife:IsA("Model") then
                if not ClonedKnife.PrimaryPart then
                    for _, Child in GetChildren(ClonedKnife) do
                        if Child:IsA("BasePart") then
                            ClonedKnife.PrimaryPart = Child
                            break
                        end
                    end
                end

                if ClonedKnife.PrimaryPart then
                    for _, Descendant in ClonedKnife:GetDescendants() do
                        if Descendant:IsA("BasePart") then
                            Descendant.CanCollide = false
                            Descendant.Massless = true
                            Descendant.Anchored = false

                            local Weld = Utility:CreateInstances("Weld", {
                                Part0 = HandleR,
                                Part1 = Descendant,
                                C0 = Offset,
                                C1 = Descendant.CFrame:ToObjectSpace(ClonedKnife.PrimaryPart.CFrame),
                                Parent = Descendant
                            })

                            TableInsert(State.Welds, Weld)

                        elseif Descendant:IsA("ParticleEmitter") or Descendant:IsA("Beam") or Descendant:IsA("Trail") then
                            Descendant.Enable = true
                        end
                    end
                end

                ClonedKnife.Parent = DefaultPart

            elseif ClonedKnife:IsA("BasePart") then
                ClonedKnife.CanCollide = false
                ClonedKnife.Massless = true
                ClonedKnife.Anchored = false

                if ClonedKnife:IsA("MeshPart") and MeleeConfig.TextureId then
                    ClonedKnife.TextureID = MeleeConfig.TextureId
                end

                ClonedKnife.Parent = DefaultPart

                local Weld = Utility:CreateInstances("Weld", {
                    Part0 = HandleR,
                    Part1 = ClonedKnife,
                    C0 = Offset,
                    Parent = ClonedKnife
                })

                TableInsert(State.Welds, Weld)
            end

            if MeleeConfig.AnimationId and MeleeConfig.AnimationId ~= "" then
                local Animator = FindFirstChildOfClass(Humanoid, "Animator")

                if Animator then
                    local Animation = Utility:CreateInstances("Animation", {
                        AnimationId = MeleeConfig.AnimationId
                    })

                    local AnimationTrack = Animator:LoadAnimation(Animation)
                    AnimationTrack.Looped = false
                    AnimationTrack:Play()

                    State.AnimationTrack = AnimationTrack
                end
            end

            if MeleeConfig.SoundId and MeleeConfig.SoundId ~= "" then
                local Sound = Utility:CreateInstances("Sound", {
                    SoundId = MeleeConfig.SoundId,
                    Parent = FindFirstChild(Tool, "Handle") or Workspace
                })

                Sound:Play()
                TableInsert(State.Sounds, Sound)
                Debris:AddItem(Sound, Max(Sound.TimeLength, 3))
            end

            Tool:SetAttribute("SkinName", SkinName)
        end

        function SkinChanger:Process(ToolName, SkinName)
            local SkinConfig = shared.prada.Character["Skin-Changer"]

            if not SkinConfig.Enabled then
                return
            end

            if ToolName == "[Knife]" then
                local Tool = self:ResolveTool("[Knife]")

                if not Tool then
                    return
                end

                if not SkinName or SkinName == "" then
                    self:RevertMelee(Tool)
                    return
                end

                local PlayerCharacter = Character:GetCharacters(LocalPlayer)

                if Tool.Parent == PlayerCharacter then
                    self:ApplyMelee(PlayerCharacter, Tool, SkinName)
                end
            else
                if not SkinName or SkinName == "" then
                    self:RevertFirearm(ToolName)
                    return
                end

                self:ApplyFirearm(ToolName, SkinName)
            end
        end

        function SkinChanger:Refresh()
            local SkinConfig = shared.prada.Character["Skin-Changer"]

            if not SkinConfig.Enabled then
                return
            end

            local Skins = SkinConfig.Data

            for ToolName, SkinName in Skins do
                self:Process(ToolName, SkinName)
            end
        end
    end

            local function CastBulletRay(Origin, Direction, Params, Range, Penetrate)
                local CurrentOrigin = Origin
                local DistanceLeft = Range
                local WallsLeft = Penetrate or 0
                local Result
                
                for _ = 1, 5 do
                    Result = Workspace:Raycast(CurrentOrigin, Direction * DistanceLeft, Params)
                    if not Result then return nil end
                    
                    local Inst = Result.Instance
                    if not Inst then return Result end
                    
                    local HitModel = Inst:FindFirstAncestorOfClass("Model")
                    local IsChar = HitModel and HitModel:FindFirstChildOfClass("Humanoid") ~= nil

                    -- Wall-Bang: keep punching through solid walls while penetration remains.
                    -- We only "land" the bullet on a character (or when we run out of walls).
                    if not IsChar and (Inst.CanCollide and Inst.Transparency < 1) and WallsLeft > 0 then
                        WallsLeft = WallsLeft - 1
                        local Dist = (Result.Position - CurrentOrigin).Magnitude
                        CurrentOrigin = Result.Position + (Direction * 0.01)
                        DistanceLeft = DistanceLeft - Dist
                        TableInsert(Params.FilterDescendantsInstances, Inst)
                        if DistanceLeft <= 0 then
                            return nil
                        end
                        continue
                    end

                    if IsChar or (Inst.CanCollide and Inst.Transparency < 1) then
                        return Result
                    end
                    

                    local Dist = (Result.Position - CurrentOrigin).Magnitude
                    CurrentOrigin = Result.Position + (Direction * 0.01)
                    DistanceLeft = DistanceLeft - Dist
                    
                    if DistanceLeft <= 0 then
                        return nil
                    end
                    

                    TableInsert(Params.FilterDescendantsInstances, Inst)
                end
                
                return Result
            end

            local function Shoot(Arguments)
                local Shooter = Arguments.Shooter
                local Handle = Arguments.Handle
                local AimPosition = Arguments.AimPosition
                local VisualAimPosition = Arguments.VisualAimPosition or AimPosition
                local BeamColor = Arguments.BeamColor
                local isReflecting = Arguments.isReflecting
                local Hit = Arguments.Hit
                local WeaponRange = Arguments.Range or 200
                local SuppressVisuals = Arguments.SuppressVisuals
                local Penetration = Arguments.Penetration or 0
                
                local SkinName = nil
                if Handle then
                    SkinName = Handle:GetAttribute("SkinName")
                end

                local IsLeftHand = Arguments.IsLeftHand
                local PlayerFromChar = Players:GetPlayerFromCharacter(Shooter)
                local HasGunFX = false

                if PlayerFromChar then
                    HasGunFX = PlayerFromChar:GetAttribute("GunFX") == true
                end

                local _, MaxDistance = GetAim(Handle.Position, WeaponRange)

                if Shooter ~= LocalPlayer.Character then
                    MaxDistance = WeaponRange
                end

                local MuzzleOrigin = Arguments.ForcedOrigin

                if not MuzzleOrigin then
                    MuzzleOrigin = GetMuzzlePosition(Handle.Parent, Handle)
                end

                local RayParams = RaycastParams.new()
                local FilterInstances = { Shooter }
                
                local IgnoredFolder = Workspace:FindFirstChild("Ignored")
                if IgnoredFolder then
                    table.insert(FilterInstances, IgnoredFolder)
                end
                
                local IgnoredModule = require(ReplicatedStorage.MainModule).Ignored
                if type(IgnoredModule) == "table" then
                    for _, inst in ipairs(IgnoredModule) do
                        table.insert(FilterInstances, inst)
                    end
                end
                
                RayParams.FilterDescendantsInstances = FilterInstances
                RayParams.FilterType = Enum.RaycastFilterType.Exclude
                RayParams.IgnoreWater = true

                -- FIXED: Give the Server Ray its own separate RaycastParams
                local ServerRayParams = RaycastParams.new()
                ServerRayParams.FilterDescendantsInstances = table.clone(FilterInstances)
                ServerRayParams.FilterType = Enum.RaycastFilterType.Exclude
                ServerRayParams.IgnoreWater = true

                local VisualHitInstance, VisualHitPosition, VisualHitNormal = nil, nil, nil
                local ServerHitInstance, ServerHitPosition, ServerHitNormal = nil, nil, nil

                if not SuppressVisuals then
                    local VisualShotDirection = (VisualAimPosition - MuzzleOrigin).Unit
                    local VisualRayResult = CastBulletRay(MuzzleOrigin, VisualShotDirection, RayParams, WeaponRange)

                    if VisualRayResult then
                        VisualHitInstance = VisualRayResult.Instance
                        VisualHitPosition = VisualRayResult.Position
                        VisualHitNormal = VisualRayResult.Normal
                    else
                        VisualHitPosition = MuzzleOrigin + VisualShotDirection * math.min(MaxDistance, WeaponRange)
                    end
                end

                local ServerShotDirection = (AimPosition - MuzzleOrigin).Unit
                local ServerRayResult = CastBulletRay(MuzzleOrigin, ServerShotDirection, ServerRayParams, WeaponRange, Penetration)

                if ServerRayResult then
                    ServerHitInstance = ServerRayResult.Instance
                    ServerHitPosition = ServerRayResult.Position
                    ServerHitNormal = ServerRayResult.Normal
                else
                    ServerHitPosition = MuzzleOrigin + ServerShotDirection * math.min(MaxDistance, WeaponRange)
                end

                if not SuppressVisuals then
                    local Part = Utility:CreateInstances("Part", {
                        Name = "BULLET_RAYS",
                        Anchored = true,
                        CanCollide = false,
                        Size = Vector3Zero,
                        Transparency = 1,
                        CFrame = CFrameNew(MuzzleOrigin, VisualHitPosition),
                        Material = Enum.Material.SmoothPlastic,
                        Parent = Workspace.Ignored.Siren.Radius,
                    })

                    Part:SetAttribute("OwnerCharacter", Shooter.Name)
                    FindServices.Debris:AddItem(Part, 1)

                    local Attachment = Utility:CreateInstances("Attachment", {
                        Position = Vector3Zero,
                        Parent = Part,
                    })

                    local Attachment2 = Utility:CreateInstances("Attachment", {
                        Position = Vector3New(0, 0, -(VisualHitPosition - MuzzleOrigin).Magnitude),
                        Parent = Part,
                    })

                    local HasMuzzleParticle = false
                    local CustomBeamPart = nil
                    local NewBeam

                    if Handle then
                        local WeaponName = Handle.Parent and Handle.Parent.Name

                        if WeaponName and not HasGunFX and (SkinName and SkinName ~= "" or SkinAssets.GunSkinMuzzleParticle:FindFirstChild(WeaponName)) then
                            if SkinName == "" then
                                SkinName = WeaponName
                            else
                                SkinName = SkinName or WeaponName
                            end

                            local MuzzleNameStr = IsLeftHand and "LeftMuzzle" or "Muzzle"
                            local ActualSkinName = FindFirstChild(SkinAssets.GunSkinMuzzleParticle, SkinName)

                            if ActualSkinName then
                                if not isReflecting then
                                    local MuzzleFolder = FindFirstChild(SkinAssets.GunSkinMuzzleParticle[SkinName], MuzzleNameStr)
                                    local DefaultMuzzle = FindFirstChild(Handle.Parent, "Default")
                                    local MeshPart = DefaultMuzzle and FindFirstChild(DefaultMuzzle, "Mesh")

                                    local MuzzleAtt = MeshPart and FindFirstChild(MeshPart, MuzzleNameStr) or FindFirstChildOfClass(Handle, MuzzleNameStr)
                                    local IsLeftHandOverride = false

                                    if IsLeftHand then
                                        local LeftMeshMuzzle = nil
                                        if MeshPart then
                                            LeftMeshMuzzle = FindFirstChild(MeshPart.DualWieldLeftHandMesh, MuzzleNameStr) or MuzzleAtt
                                        else
                                            LeftMeshMuzzle = MuzzleAtt
                                        end
                                        
                                        IsLeftHandOverride = LeftMeshMuzzle ~= MuzzleAtt
                                        if IsLeftHandOverride then
                                            MuzzleAtt = LeftMeshMuzzle
                                        end
                                    end

                                    if MuzzleAtt then
                                        local ParticleSource
                                        if IsLeftHandOverride then
                                            ParticleSource = MuzzleAtt
                                        elseif SkinAssets.GunSkinMuzzleParticle[SkinName].Muzzle:FindFirstChild("Different_GunMuzzle") then
                                            ParticleSource = SkinAssets.GunSkinMuzzleParticle[SkinName][MuzzleNameStr].Different_GunMuzzle[WeaponName]
                                        else
                                            ParticleSource = SkinAssets.GunSkinMuzzleParticle[SkinName][MuzzleNameStr]
                                        end

                                        for _, Child in ipairs(GetChildren(ParticleSource)) do
                                            local EmitCount = Child:GetAttribute("EmitCount") or 1
                                            local ClonedParticle = IsLeftHand and Child or Child:Clone()
                                            ClonedParticle.Parent = MuzzleAtt
                                            ClonedParticle:Emit(EmitCount)

                                            if not IsLeftHandOverride then
                                                TaskDelay(ClonedParticle.Lifetime.Max, function() 
                                                    ClonedParticle:Destroy() 
                                                end)
                                            end
                                        end
                                    end
                                else
                                    local RandomParticles = GetChildren(SkinAssets.GunSkinMuzzleParticle[SkinName])
                                    local ChosenParticle = RandomParticles[math.random(#RandomParticles)]:Clone()
                                    ChosenParticle.Parent = Attachment
                                    ChosenParticle:Emit(ChosenParticle.Rate)
                                end

                                HasMuzzleParticle = true
                            end

                            if SkinAssets.GunBeam:FindFirstChild(SkinName) then
                                local BeamFolder = SkinAssets.GunBeam[SkinName]
                                local BeamNameStr = IsLeftHand and "LeftGunBeam" or "GunBeam"
                                local BeamInstance = FindFirstChild(BeamFolder, BeamNameStr) or BeamFolder.GunBeam

                                if BeamInstance:IsA("BasePart") then
                                    NewBeam = { Parent = nil, Attachment0 = nil, Attachment1 = nil }
                                    local DifferentBeam = FindFirstChild(BeamInstance, "Different_GunBeam")

                                    if DifferentBeam and FindFirstChild(DifferentBeam, WeaponName) then
                                        local CustomBeam = DifferentBeam[WeaponName][BeamNameStr]
                                        if CustomBeam and CustomBeam:IsA("BasePart") then
                                            CustomBeamPart = CustomBeam:Clone()
                                        elseif CustomBeam then
                                            NewBeam = CustomBeam:Clone() or NewBeam
                                        end
                                    else
                                        CustomBeamPart = BeamInstance:Clone()
                                    end
                                else
                                    NewBeam = BeamInstance:Clone()
                                end
                            else
                                NewBeam = ReplicatedStorage.GunBeam:Clone()
                                NewBeam.Color = BeamColor and ColorSequence.new(BeamColor) or NewBeam.Color
                            end
                        else
                            NewBeam = ReplicatedStorage.GunBeam:Clone()
                            NewBeam.Color = BeamColor and ColorSequence.new(BeamColor) or NewBeam.Color
                        end
                    else
                        NewBeam = nil
                    end

                    TaskSpawn(function()
                        if CustomBeamPart then
                            local Distance = (VisualHitPosition - MuzzleOrigin).Magnitude
                            local TweenTime = Distance / 725
                            
                            CustomBeamPart.Anchored = true
                            CustomBeamPart.CanCollide = false
                            CustomBeamPart.CanQuery = false
                            CustomBeamPart.CFrame = CFrameNew(MuzzleOrigin, VisualHitPosition)
                            
                            local EndCFrame = CustomBeamPart.CFrame * CFrameNew(0, 0, -Distance)
                            CustomBeamPart.Parent = Workspace.Ignored.Siren.Radius
                            
                            TaskDelay(TweenTime + 5, function() 
                                if CustomBeamPart then 
                                    CustomBeamPart:Destroy() 
                                end 
                            end)

                            if CustomBeamPart:GetAttribute("SpecialEffects") then
                                for _, Descendant in ipairs(GetDescendants(CustomBeamPart)) do
                                    if Descendant:IsA("Trail") and Descendant:GetAttribute("ColorRandom") then
                                        local ColorSeq = Descendant:GetAttribute("ColorRandom")
                                        Descendant.Color = ColorSequence.new(EvaluateColorSequence(ColorSeq, math.random()))
                                    end
                                end
                            end

                            local Tween1 = TweenService:Create(CustomBeamPart, TweenInfo.new(0.05, Enum.EasingStyle.Linear), { 
                                CFrame = CustomBeamPart.CFrame * CFrameNew(0, 0, -0.1) 
                            })
                            Tween1:Play()
                            TaskWait(0.05)

                            if Tween1.PlaybackState ~= Enum.PlaybackState.Completed then
                                Tween1:Pause()
                            end

                            local Tween2 = nil
                            if _G.Reduce_Lag and not CustomBeamPart:GetAttribute("NoSlow") or CustomBeamPart:GetAttribute("LOWGFX") then
                                CustomBeamPart.CFrame = EndCFrame
                            else
                                Tween2 = TweenService:Create(CustomBeamPart, TweenInfo.new(TweenTime, Enum.EasingStyle.Linear), { 
                                    CFrame = EndCFrame 
                                })
                                Tween2:Play()
                                TaskWait(TweenTime)
                            end

                            if FindFirstChild(CustomBeamPart, "Impact") and VisualHitInstance and VisualHitNormal and VisualHitInstance.Parent and not FindFirstChild(VisualHitInstance.Parent, "Humanoid") then
                                if Tween2 and Tween2.PlaybackState ~= Enum.PlaybackState.Completed then
                                    TaskWait(0.05)
                                end

                                if not FindFirstChild(CustomBeamPart, "NoNormal") then
                                    CustomBeamPart.CFrame = CFrameNew(VisualHitPosition, VisualHitPosition - VisualHitNormal)
                                end

                                for _, Child in ipairs(GetChildren(CustomBeamPart.Impact)) do
                                    if Child:IsA("ParticleEmitter") then
                                        Child:Emit(Child:GetAttribute("EmitCount") or 1)
                                    end
                                end
                            else
                                for _, Child in ipairs(GetChildren(CustomBeamPart)) do
                                    if Child:IsA("BasePart") then
                                        Child.Transparency = 1
                                    end
                                end
                            end

                            if CustomBeamPart then
                                for _, Descendant in ipairs(GetDescendants(CustomBeamPart)) do
                                    if Descendant:IsA("ParticleEmitter") then
                                        Descendant.Enabled = false
                                    end
                                end
                            end
                        elseif VisualHitInstance and VisualHitNormal and VisualHitInstance:IsDescendantOf(Workspace.MAP) and not HasGunFX and SkinName and SkinAssets.GunBeam:FindFirstChild(SkinName) and SkinAssets.GunBeam[SkinName]:FindFirstChild("Impact") then
                            local ImpactPart = SkinAssets.GunBeam[SkinName].Impact:Clone()
                            ImpactPart.Parent = Workspace.Ignored
                            ImpactPart:PivotTo(CFrameNew(VisualHitPosition, VisualHitPosition + VisualHitNormal * 5) * CFrame.Angles(-1.5707963267948966, 0, 0))
                            
                            for _, Descendant in ipairs(GetDescendants(ImpactPart)) do
                                if Descendant:IsA("ParticleEmitter") then
                                    Descendant:Emit(Descendant:GetAttribute("EmitCount") or 1)
                                end
                            end

                            TaskDelay(1.5, function() 
                                if ImpactPart then 
                                    ImpactPart:Destroy() 
                                end 
                            end)
                        end

                        local PointLight = Utility:CreateInstances("PointLight", {
                            Brightness = 0.5,
                            Range = 15,
                            Shadows = true,
                            Color = Color3.new(1, 1, 1),
                            Parent = Part,
                        })

                        local ShootBBGUI = FindFirstChild(Handle, "ShootBBGUI")
                        local BBGUIElement = ShootBBGUI and (not HasMuzzleParticle and FindFirstChild(ShootBBGUI, "Shoot"))

                        if BBGUIElement then
                            BBGUIElement.Size = UDim2.new(0, 0, 0, 0)
                            BBGUIElement.ImageTransparency = 1
                            BBGUIElement.Visible = true
                            
                            TweenService:Create(BBGUIElement, TweenInfo.new(0.4, Enum.EasingStyle.Bounce, Enum.EasingDirection.In, 0, false, 0), { 
                                ImageTransparency = 0.4, 
                                Size = UDim2.new(1, 0, 1, 0) 
                            }):Play()
                            
                            TweenService:Create(PointLight, TweenInfo.new(0.4, Enum.EasingStyle.Bounce, Enum.EasingDirection.In, 0, false, 0), { 
                                Range = 0 
                            }):Play()
                            
                            TaskWait(0.4)
                            Part:Destroy()
                            
                            TweenService:Create(BBGUIElement, TweenInfo.new(0.2, Enum.EasingStyle.Bounce, Enum.EasingDirection.In, 0, false, 0), { 
                                ImageTransparency = 1, 
                                Size = UDim2.new(1, 0, 1, 0) 
                            }):Play()
                            
                            TaskWait(0.2)
                            BBGUIElement.Visible = false
                        end
                    end)

                    NewBeam.Attachment0 = Attachment
                    NewBeam.Attachment1 = Attachment2
                    NewBeam.Name = "NewGunBeam"
                    NewBeam.Parent = Part
                end

                if Shooter == LocalPlayer.Character and Workspace:GetServerTimeNow() - ServerTime > 0.95 then
                    PlayerShot(Handle)
                end

                if not SoundDebounce[Handle] then
                    if not Handle.Parent then
                        return MuzzleOrigin, nil, nil, nil
                    end

                    local SoundSkin = SkinName or "None"
                    local IsAllSound = SoundSequences[SoundSkin] ~= "All" and not (SoundSequences[SoundSkin] and SoundSequences[SoundSkin][Handle.Parent.Name])
                    
                    TaskSpawn(GunSoundPlay.SoundPlay, Handle, IsAllSound and true or false, HasGunFX)
                    SoundDebounce[Handle] = true
                    
                    TaskDelay(0.021, function() 
                        SoundDebounce[Handle] = nil 
                    end)
                end

                if not SuppressVisuals then
                    TaskSpawn(function()
                        if Handle:GetAttribute("DualWield") and not IsLeftHand then
                            local Default = FindFirstChild(Handle.Parent, "Default")
                            local Mesh = Default and FindFirstChild(Default, "Mesh")
                            local LeftMesh = Mesh and FindFirstChild(Mesh, "DualWieldLeftHandMesh")
                            local LeftMuzzle = LeftMesh and FindFirstChild(LeftMesh, "LeftMuzzle")
                            
                            if LeftMuzzle then
                                Shoot({
                                    IsLeftHand = true,
                                    Shooter = Shooter,
                                    Handle = Handle,
                                    AimPosition = VisualHitPosition,
                                    VisualAimPosition = VisualHitPosition,
                                    BeamColor = BeamColor,
                                    Hit = Hit,
                                    Range = WeaponRange
                                })
                            end
                        end
                    end)
                end

                return MuzzleOrigin, ServerHitPosition, ServerHitInstance, ServerHitNormal
            end

            local function GetRageHitPartName(Mode)
                if Mode == "Max" or Mode == "Full" then
                    return "Head"
                elseif Mode == "Half" or Mode == "Partial" then
                    return "HumanoidRootPart"
                elseif Mode == "Min" or Mode == "Minimal" then
                    return "RightUpperArm"
                end
                
                return "Head"
            end

            local function GetSpreadVector(SpreadMultiplier)
                local SpreadVectorX = (math.random() > 0.5 and math.random() * 0.05 or -math.random() * 0.05) * SpreadMultiplier
                local SpreadVectorY = (math.random() > 0.5 and math.random() * 0.1 or -math.random() * 0.1) * SpreadMultiplier
                local SpreadVectorZ = (math.random() > 0.5 and math.random() * 0.05 or -math.random() * 0.05) * SpreadMultiplier
                return Vector3New(SpreadVectorX, SpreadVectorY, SpreadVectorZ)
            end

            local function GetMouseHitPosition(MuzzlePosition, WeaponRange, SpreadMultiplier)
                local MouseAimDirection = GetAim(MuzzlePosition, WeaponRange)
                local CurrentSpreadVector = GetSpreadVector(SpreadMultiplier)
                return MuzzlePosition + (MouseAimDirection + CurrentSpreadVector) * WeaponRange
            end

            local function GetSilentAimPosition(ActiveTarget, SpreadMultiplier, MuzzlePosition)
                if not ActiveTarget then
                    return nil
                end

                local PerfectPos = Assist:SilentAimPosition(ActiveTarget)
                if not PerfectPos then return nil end

                if SpreadMultiplier and SpreadMultiplier > 0 then
                    local Direction = PerfectPos - MuzzlePosition
                    local Distance = Direction.Magnitude
                    local UnitDirection = Direction.Unit
                    
                    local Spread = GetSpreadVector(SpreadMultiplier)
                    return MuzzlePosition + (UnitDirection + Spread) * Distance
                end

                return PerfectPos
            end

            local function GetDamageOverridePosition(ActiveTarget, OverrideMode, MuzzlePosition, WeaponRange, SpreadMultiplier)
                if not ActiveTarget or not OverrideMode or OverrideMode == "None" then
                    return nil
                end

                local OverridePartName = GetRageHitPartName(OverrideMode)
                local TargetAimDirection, Distance = GetAim(MuzzlePosition, WeaponRange, ActiveTarget, OverridePartName)
                local PerfectPos = MuzzlePosition + TargetAimDirection * Distance

                if SpreadMultiplier and SpreadMultiplier > 0 then
                    local Spread = GetSpreadVector(SpreadMultiplier)
                    return MuzzlePosition + (TargetAimDirection + Spread) * Distance
                end

                return PerfectPos
            end

            local function SetupTool(ToolParent)
                if not ToolParent or ToolParent.Name == "[Knife]" then
                    return
                end

                if ToolParent:GetAttribute("zzzSetup") then
                    return
                end
                
                ToolParent:SetAttribute("zzzSetup", true)

                for _, Descendant in ipairs(GetDescendants(ToolParent)) do
                    if Descendant:IsA("LocalScript") then
                        Descendant:Destroy()
                    end
                end

                local Handle = ToolParent:WaitForChild("Handle")
                local PlayerCharacter = Character:GetCharacters(LocalPlayer)
                local ToolAmmo = game.PlaceId == 88976059384565 and { Value = 5 } or ToolParent:WaitForChild("Ammo", 5)
                
                if not ToolAmmo then
                    return
                end

                local WeaponRange = ToolParent:WaitForChild("Range").Value
                local OriginalCooldown = ToolParent:WaitForChild("ShootingCooldown").Value
                local ToolRemoteEvent = FindFirstChild(ToolParent, "RemoteEvent") or { FireServer = function() end }

                local LastShotTime = 0
                local IsShotgunWeapon = (ToolParent.Name == "[Double-Barrel SG]" or ToolParent.Name == "[TacticalShotgun]" or ToolParent.Name == "[Shotgun]" or ToolParent.Name == "[Drum-Shotgun]")
                local IsAutomaticWeapon = AutomaticWeapons[ToolParent.Name] == true

                ToolParent.Activated:Connect(function()
                    local SilentAimConfig = shared.prada.Combat.Silent
                    local GunMods = shared.prada.Combat.Triggerbot["Gun-Modifications"]
                    local CooldownConfig = nil
                    local BulletSpreadConfig = nil

                    local function GunListEnabled(WhitelistTable)
                        -- Every weapon is always whitelisted (Deagle + all guns).
                        return true
                    end

                    -- Range-Enhancer: recompute the effective shot range at fire-time from prada config.
                    -- This extends BOTH the client aim helpers and the Shoot() raycast (the server trusts
                    -- the client-computed hit position), so the bullet genuinely reaches beyond the gun's
                    -- natural Range NumberValue. Per-gun Data wins, else the shared Override, else base.
                    local EffectiveRange = WeaponRange
                    local RangeCheck = SilentAimConfig["Range-Check"]
                    if RangeCheck and RangeCheck.Enabled and RangeCheck["Range-Enhancer"] then
                        local Configured = RangeCheck.Data and RangeCheck.Data[ToolParent.Name]
                        if Configured and Configured ~= 0 then
                            EffectiveRange = Configured
                        elseif RangeCheck.Override and RangeCheck.Override ~= 0 then
                            EffectiveRange = RangeCheck.Override
                        end
                    end

                    -- Wall-Bang: pick a random wall-penetration depth in [Values[1], Values[2]].
                    local PenetrationWalls = 0
                    local WallBang = SilentAimConfig["Wall-Bang"]
                    if WallBang and WallBang.Enabled then
                        local WMin, WMax = WallBang.Settings.Values[1], WallBang.Settings.Values[2]
                        PenetrationWalls = WMax > WMin and math.random(WMin, WMax) or WMin
                    end

                    -- Whitelist gating: if a feature whitelists this gun as false, disable it for this shot.
                    local SilentAllowed = GunListEnabled(SilentAimConfig.Whitelisted)
                    local CamlockEnabled = shared.prada.Combat.Camlock.Enabled
                    local TriggerbotEnabled = shared.prada.Combat.Triggerbot.Enabled
                    local TriggerAllowed = GunListEnabled(shared.prada.Combat.Triggerbot.Whitelisted)
                    local SilentClientRedir = SilentAimConfig["Client Redirection"]
                    local ClientRedirAllowed = SilentClientRedir.Enabled and GunListEnabled(SilentClientRedir.Whitelisted) and SilentAllowed

                    local CooldownModifier = 100
                    if CooldownConfig and CooldownConfig['Enable'] and CooldownConfig[1] and CooldownConfig[1][ToolParent.Name] then
                        CooldownModifier = CooldownConfig[1][ToolParent.Name]['Values'] or 100
                    end
                    
                    if CooldownModifier < 0 then
                        CooldownModifier = 0
                    end

                    local CalculatedCooldown = OriginalCooldown * (CooldownModifier / 100)
                    local BaseCooldown = CalculatedCooldown + 0.0095

                    if tick() - LastShotTime < BaseCooldown then
                        return
                    end

                    if not CanShoot(PlayerCharacter) then
                        return
                    end

                    LastShotTime = tick()

                    local SpreadMultiplier = 0
                    
                    if IsShotgunWeapon then
                        SpreadMultiplier = 1
                        if BulletSpreadConfig and BulletSpreadConfig['Enable'] and BulletSpreadConfig[1] and BulletSpreadConfig[1]['Settings'] and BulletSpreadConfig[1]['Settings'][ToolParent.Name] then
                            SpreadMultiplier = BulletSpreadConfig[1]['Settings'][ToolParent.Name]['Values'] / 100
                        end
                    end

                    local CurrentTarget = nil
                    local DamageOverrideMode = nil
                    local IsForceHitEnabled = false

                    if SilentAimConfig.Enabled and SilentAllowed then
                        CurrentTarget = Script.Status.SilentAim.ActiveTarget
                    end

                    if GunMods then
                        local ForceHitState = GunMods["Force-Hit"] and GunMods["Force-Hit"][ToolParent.Name]
                        local DamageMod = GunMods["Damage-Mod"] or {}
                        local DamageOverrideSetting = DamageMod.Data and DamageMod.Data[ToolParent.Name]

                        if ForceHitState and CurrentTarget then
                            IsForceHitEnabled = true
                        end

                        if DamageOverrideSetting and DamageOverrideSetting ~= "" and DamageOverrideSetting ~= "None" and CurrentTarget then
                            DamageOverrideMode = DamageOverrideSetting
                        end
                    end

                    local MuzzleWorldPosition = GetMuzzlePosition(ToolParent, Handle)

                    if IsShotgunWeapon then
                        ToolRemoteEvent:FireServer("Shoot")
                        local ServerTime = Workspace:GetServerTimeNow()
                        
                        for _ = 1, 5 do
                            local MouseHitPos = GetMouseHitPosition(MuzzleWorldPosition, EffectiveRange, SpreadMultiplier)
                            local SilentAimPos = GetSilentAimPosition(CurrentTarget, SpreadMultiplier, MuzzleWorldPosition)
                            local DamageOverridePos = GetDamageOverridePosition(CurrentTarget, DamageOverrideMode, MuzzleWorldPosition, EffectiveRange, SpreadMultiplier)

                            local ServerTargetPosition = DamageOverridePos or SilentAimPos or MouseHitPos
                            local VisualHitPosition = nil
                            
                            if IsForceHitEnabled then
                                VisualHitPosition = nil
                            elseif ClientRedirAllowed and not DamageOverrideMode then
                                VisualHitPosition = MouseHitPos
                            else
                                VisualHitPosition = SilentAimPos or MouseHitPos
                            end

                            local MuzzlePosition, HitPosition, HitInstance, HitNormalVector = Shoot({
                                Shooter = PlayerCharacter,
                                Handle = Handle,
                                AimPosition = ServerTargetPosition,
                                VisualAimPosition = VisualHitPosition,
                                BeamColor = Color3.new(1, 0.545098, 0.14902),
                                    Range = EffectiveRange,
                                    Penetration = PenetrationWalls,
                                    SuppressVisuals = IsForceHitEnabled
                            })

                            if SilentAimConfig.ClientAmmoSaver and not IsForceHitEnabled then
                                local HitModel = HitInstance and HitInstance:FindFirstAncestorOfClass("Model")
                                local IsCharacter = HitModel and FindFirstChildOfClass(HitModel, "Humanoid") ~= nil
                                
                                if not IsCharacter then
                                    continue
                                end
                            end

                            MainEvent:FireServer("ShootGun", Handle, MuzzlePosition, HitPosition, HitInstance, HitNormalVector, ServerTime)
                        end
                        
                        ToolRemoteEvent:FireServer()
                    elseif IsAutomaticWeapon then
                        ToolRemoteEvent:FireServer("Shoot")
                        local IsShooting = true
                        
                        TaskSpawn(function()
                            while IsShooting and ToolParent.Parent == PlayerCharacter and ToolAmmo.Value > 0 do
                                local MouseHitPos = GetMouseHitPosition(MuzzleWorldPosition, EffectiveRange, SpreadMultiplier)
                                local SilentAimPos = GetSilentAimPosition(CurrentTarget, SpreadMultiplier, MuzzleWorldPosition)
                                local DamageOverridePos = GetDamageOverridePosition(CurrentTarget, DamageOverrideMode, MuzzleWorldPosition, EffectiveRange, SpreadMultiplier)

                                local ServerTargetPosition = DamageOverridePos or SilentAimPos or MouseHitPos
                                local VisualHitPosition = nil
                                
                                if IsForceHitEnabled then
                                    VisualHitPosition = nil
                                elseif ClientRedirAllowed and not DamageOverrideMode then
                                    VisualHitPosition = MouseHitPos
                                else
                                    VisualHitPosition = SilentAimPos or MouseHitPos
                                end

                                local MuzzlePosition, HitPosition, HitInstance, HitNormalVector = Shoot({
                                    Shooter = PlayerCharacter,
                                    Handle = Handle,
                                    AimPosition = ServerTargetPosition,
                                    VisualAimPosition = VisualHitPosition,
                                    BeamColor = Color3.new(1, 0.545098, 0.14902),
                                    Range = EffectiveRange,
                                    Penetration = PenetrationWalls,
                                    SuppressVisuals = IsForceHitEnabled
                                })

                                if SilentAimConfig.ClientAmmoSaver and not IsForceHitEnabled then
                                    local HitModel = HitInstance and HitInstance:FindFirstAncestorOfClass("Model")
                                    local IsCharacter = HitModel and FindFirstChildOfClass(HitModel, "Humanoid") ~= nil
                                    
                                    if not IsCharacter then
                                        IsShooting = false
                                        break
                                    end
                                end

                                MainEvent:FireServer("ShootGun", Handle, MuzzlePosition, HitPosition, HitInstance, HitNormalVector)
                                TaskWait(CalculatedCooldown + 0.0095)
                            end
                        end)
                        
                        ToolParent.Deactivated:Wait()
                        IsShooting = false
                        ToolRemoteEvent:FireServer()
                    else
                        ToolRemoteEvent:FireServer("Shoot")
                        
                        local MouseHitPos = GetMouseHitPosition(MuzzleWorldPosition, EffectiveRange, SpreadMultiplier)
                        local SilentAimPos = GetSilentAimPosition(CurrentTarget, SpreadMultiplier, MuzzleWorldPosition)
                        local DamageOverridePos = GetDamageOverridePosition(CurrentTarget, DamageOverrideMode, MuzzleWorldPosition, EffectiveRange, SpreadMultiplier)

                        local ServerTargetPosition = DamageOverridePos or SilentAimPos or MouseHitPos
                        local VisualHitPosition = nil
                        
                        if IsForceHitEnabled then
                            VisualHitPosition = nil
                        elseif ClientRedirAllowed and not DamageOverrideMode then
                            VisualHitPosition = MouseHitPos
                        else
                            VisualHitPosition = SilentAimPos or MouseHitPos
                        end

                        local MuzzlePosition, HitPosition, HitInstance, HitNormalVector = Shoot({
                            Shooter = PlayerCharacter,
                            Handle = Handle,
                            AimPosition = ServerTargetPosition,
                            VisualAimPosition = VisualHitPosition,
                            BeamColor = Color3.new(1, 0.545098, 0.14902),
                            Range = EffectiveRange,
                            Penetration = PenetrationWalls,
                            SuppressVisuals = IsForceHitEnabled
                        })

                        if SilentAimConfig.ClientAmmoSaver and not IsForceHitEnabled then
                            local HitModel = HitInstance and HitInstance:FindFirstAncestorOfClass("Model")
                            local IsCharacter = HitModel and FindFirstChildOfClass(HitModel, "Humanoid") ~= nil
                            
                            if not IsCharacter then
                                ToolRemoteEvent:FireServer()
                                return
                            end
                        end

                        MainEvent:FireServer("ShootGun", Handle, MuzzlePosition, HitPosition, HitInstance, HitNormalVector)
                        ToolRemoteEvent:FireServer()
                    end
                end)
            end

            return {
                CanShoot = CanShoot,
                Animate = PlayerShot,
                GetAim = GetAim,
                ColorTransform = EvaluateColorSequence,
                ShootGun = Shoot,
                SetupTool = SetupTool
            }
        end
    end

    local function __Tool_Handling()
        local ToolConnections = {}
        local CurrentTool = nil
        local StateLoop = nil
        local AppliedState = {}

        local GunSystem = Functionalities.GunModule(Script, Closest, Assist)

        local function HandleTool(PlayerCharacter, Object)
            if not Object or not Object:IsA("Tool") then
                return
            end

            CurrentTool = Object

            AppliedState[Object] = {Skin = ""}

            if shared.prada.Character["Skin-Changer"].Enabled then
                local SkinName = shared.prada.Character["Skin-Changer"].Data[Object.Name]

                if SkinName and SkinName ~= "" then
                    if Object.Name == "[Knife]" then
                        SkinChanger:ApplyMelee(PlayerCharacter, Object, SkinName)
                    else
                        SkinChanger:ApplyFirearm(Object.Name, SkinName)
                    end

                    AppliedState[Object].Skin = SkinName
                end
            else
                if Object.Name == "[Knife]" then
                    SkinChanger:RevertMelee(Object)
                else
                    SkinChanger:RevertFirearm(Object.Name)
                end
                AppliedState[Object].Skin = ""
            end

            GunSystem.SetupTool(Object)

            if StateLoop then
                task.cancel(StateLoop)
            end

            StateLoop = task.spawn(function()
                while CurrentTool == Object and Object.Parent == PlayerCharacter do
                    task.wait(0.1)

                    local ConfigSkin = ""

                    if shared.prada.Character["Skin-Changer"].Enabled then
                        ConfigSkin = shared.prada.Character["Skin-Changer"].Data[Object.Name] or ""
                    end

                    if ConfigSkin ~= AppliedState[Object].Skin then
                        if ConfigSkin == "" then
                            if Object.Name == "[Knife]" then
                                SkinChanger:RevertMelee(Object)
                            else
                                SkinChanger:RevertFirearm(Object.Name)
                            end
                        else
                            if Object.Name == "[Knife]" then
                                SkinChanger:ApplyMelee(PlayerCharacter, Object, ConfigSkin)
                            else
                                SkinChanger:ApplyFirearm(Object.Name, ConfigSkin)
                            end
                        end

                        AppliedState[Object].Skin = ConfigSkin
                    end
                end
            end)
        end

        local function AddCharacter(PlayerCharacter)
            for _, Object in ipairs(GetChildren(PlayerCharacter)) do
                if Object:IsA("Tool") then
                    HandleTool(PlayerCharacter, Object)
                end
            end

            Utility:CreateConnections(PlayerCharacter.ChildAdded, function(Object)
                HandleTool(PlayerCharacter, Object)
            end)

            Utility:CreateConnections(PlayerCharacter.ChildRemoved, function(Object)
                if Object == CurrentTool then
                    if StateLoop then
                        task.cancel(StateLoop)
                        StateLoop = nil
                    end

                    if Object.Name == "[Knife]" then
                        SkinChanger:RevertMelee(Object)
                    else
                        SkinChanger:RevertFirearm(Object.Name)
                    end

                    for _, Connection in ipairs(ToolConnections) do
                        Connection:Disconnect()
                    end

                    TableClear(ToolConnections)
                    AppliedState[Object] = nil
                    CurrentTool = nil
                end
            end)
        end

        Utility:CreateConnections(LocalPlayer.CharacterAdded, function(PlayerCharacter)
            AddCharacter(PlayerCharacter)
        end)

        if LocalPlayer.Character then
            AddCharacter(LocalPlayer.Character)
        end
    end

                Clothing = {},
                BodyParts = {},

                Scales = {},
                Transparencies = {},
                FaceTexture = ""

                if Child:IsA("Accessory") then
                    table.insert(self.OriginalState.Accessories, Child:Clone())
                elseif Child:IsA("Shirt") or Child:IsA("Pants") or Child:IsA("ShirtGraphic") then
                    table.insert(self.OriginalState.Clothing, Child:Clone())
                end
            end
            
            for PartName, _ in BodyPartMap do
                local Part = FindFirstChild(PlayerCharacter, PartName)
                
                if Part and Part:IsA("MeshPart") then
                    self.OriginalState.BodyParts[PartName] = Part:Clone()
                end
            end

        function AvatarSpoofer:ApplyBodyParts(PlayerCharacter, Dummy)
            local Humanoid = FindFirstChildOfClass(PlayerCharacter, "Humanoid")
            
            if not Humanoid then
                return
            end

            for PartName, _ in BodyPartMap do
                local CharPart = FindFirstChild(PlayerCharacter, PartName)
                local DummyPart = FindFirstChild(Dummy, PartName)
                
                if CharPart and DummyPart and CharPart:IsA("MeshPart") and DummyPart:IsA("MeshPart") then
                    pcall(function()
                        CharPart:ApplyMesh(DummyPart)
                    end)
                    
                    pcall(function()
                        CharPart.TextureID = DummyPart.TextureID
                    end)
                    
                    pcall(function()
                        CharPart.Size = DummyPart.Size
                    end)

                    local OrigSize = FindFirstChild(DummyPart, "OriginalSize")
                    
                    if OrigSize then
                        local COrigSize = FindFirstChild(CharPart, "OriginalSize")
                        
                        if COrigSize then
                            COrigSize.Value = OrigSize.Value
                        end
                    end

                    local ScaleType = FindFirstChild(DummyPart, "AvatarPartScaleType")
                    
                    if ScaleType then
                        local CScaleType = FindFirstChild(CharPart, "AvatarPartScaleType")
                        
                        if CScaleType then
                            CScaleType.Value = ScaleType.Value
                        else
                            local ClonedScale = ScaleType:Clone()
                            ClonedScale.Parent = CharPart
                        end
                    else
                        local CScaleType = FindFirstChild(CharPart, "AvatarPartScaleType")
                        
                        if CScaleType then
                            CScaleType:Destroy()
                        end
                    end

                    for _, Child in GetChildren(CharPart) do
                        if Child:IsA("Attachment") then
                            local DummyAtt = FindFirstChild(DummyPart, Child.Name)
                            
                            if DummyAtt then
                                Child.CFrame = DummyAtt.CFrame
                                local OrigPos = FindFirstChild(DummyAtt, "OriginalPosition")
                                
                                if OrigPos then
                                    local COrigPos = FindFirstChild(Child, "OriginalPosition")
                                    
                                    if COrigPos then
                                        COrigPos.Value = OrigPos.Value
                                    end
                                end
                            end
                        end
                    end

                    for _, Child in GetChildren(CharPart) do
                        if Child:IsA("SurfaceAppearance") then
                            Child:Destroy()
                        end
                    end
                    
                    local DummySurface = FindFirstChildOfClass(DummyPart, "SurfaceAppearance")
                    
                    if DummySurface then
                        local ClonedSurface = DummySurface:Clone()
                        ClonedSurface.Parent = CharPart
                    end
                end
            end

            for _, Joint in GetDescendants(PlayerCharacter) do
                if Joint:IsA("Motor6D") then
                    local Part0 = Joint.Part0
                    local Part1 = Joint.Part1
                    
                    if Part0 and Part1 then
                        local AttName = Joint.Name .. "RigAttachment"
                        local Att0 = FindFirstChild(Part0, AttName)
                        local Att1 = FindFirstChild(Part1, AttName)
                        
                        if Att0 and Att1 then
                            Joint.C0 = Att0.CFrame
                            Joint.C1 = Att1.CFrame
                        end
                    end
                end
            end
        end

        function AvatarSpoofer:ApplyClothingAndAccessories(PlayerCharacter, SourceModel)
            local Humanoid = FindFirstChildOfClass(PlayerCharacter, "Humanoid")

            for _, Child in GetChildren(PlayerCharacter) do
                if Child:IsA("Accessory") or Child:IsA("Shirt") or Child:IsA("Pants") or Child:IsA("ShirtGraphic") then
                    Child:Destroy()
                end
            end

            for _, Child in GetChildren(SourceModel) do
                local Class = Child.ClassName
                
                if Class == "Shirt" or Class == "Pants" or Class == "ShirtGraphic" then
                    local ClonedClothing = Child:Clone()
                    ClonedClothing.Parent = PlayerCharacter
                elseif Class == "Accessory" then
                    local Clone = Child:Clone()
                    local Handle = FindFirstChild(Clone, "Handle")
                    
                    if Handle then
                        Handle.Anchored = false
                        Handle.CanCollide = false
                        
                        for _, Weld in GetChildren(Handle) do
                            if Weld:IsA("JointInstance") then
                                Weld:Destroy()
                            end
                        end

                        local HandleAtt = FindFirstChildOfClass(Handle, "Attachment")
                        
                        if HandleAtt then
                            local TargetPart = nil
                            local TargetAtt = nil
                            
                            for _, Part in GetChildren(PlayerCharacter) do
                                if Part:IsA("BasePart") then
                                    local Att = FindFirstChild(Part, HandleAtt.Name)
                                    
                                    if Att then
                                        TargetPart = Part
                                        TargetAtt = Att
                                        break
                                    end
                                end
                            end

                            if TargetPart and TargetAtt then
                                local Weld = Instance.new("Weld")
                                Weld.Name = "AccessoryWeld"
                                Weld.Part0 = Handle
                                Weld.Part1 = TargetPart
                                Weld.C0 = HandleAtt.CFrame
                                Weld.C1 = TargetAtt.CFrame
                                Weld.Parent = Handle
                                
                                Clone.Parent = PlayerCharacter
                            else
                                if Humanoid then
                                    pcall(function()
                                        Humanoid:AddAccessory(Clone)
                                    end)
                                end
                            end
                        else
                            if Humanoid then
                                pcall(function()
                                    Humanoid:AddAccessory(Clone)
                                end)
                            end
                        end
                    end
                end
            end
        end

        function AvatarSpoofer:ApplyCustomScaling(PlayerCharacter, SpoofedDescription, ScalingMode)
            local Humanoid = FindFirstChildOfClass(PlayerCharacter, "Humanoid")
            
            if not Humanoid then
                return
            end
            
            local Preset = ScalePresets[ScalingMode]
            
            for _, ScaleInfo in ScaleMap do
                local ScaleName = ScaleInfo[1]
                local DescProperty = ScaleInfo[2]
                
                local ScaleValue = FindFirstChild(Humanoid, ScaleName)
                
                if not ScaleValue then
                    ScaleValue = Instance.new("NumberValue")
                    ScaleValue.Name = ScaleName
                    ScaleValue.Parent = Humanoid
                end
                
                local NewValue = nil
                
                if Preset then
                    NewValue = Preset[ScaleName]
                elseif ScalingMode == "Target" and SpoofedDescription then
                    NewValue = SpoofedDescription[DescProperty]
                end
                
                if NewValue and NewValue > 0 then
                    pcall(function()
                        ScaleValue.Value = NewValue
                    end)
                end
            end
        end

        function AvatarSpoofer:Revert(PlayerCharacter)
            if not self.OriginalState then
                return
            end
            
            local Humanoid = FindFirstChildOfClass(PlayerCharacter, "Humanoid")
            
            for _, Child in GetChildren(PlayerCharacter) do
                if Child:IsA("Accessory") or Child:IsA("Shirt") or Child:IsA("Pants") or Child:IsA("ShirtGraphic") then
                    Child:Destroy()
                end
            end
            
            if Humanoid then
                for PartName, _ in BodyPartMap do
                    local OriginalPart = self.OriginalState.BodyParts[PartName]
                    local CharPart = FindFirstChild(PlayerCharacter, PartName)
                    
                    if OriginalPart and CharPart then
                        pcall(function()
                            CharPart:ApplyMesh(OriginalPart)
                        end)
                        
                        pcall(function()
                            CharPart.TextureID = OriginalPart.TextureID
                        end)
                        
                        pcall(function()
                            CharPart.Size = OriginalPart.Size
                        end)
                        
                        for _, Child in GetChildren(CharPart) do
                            if Child:IsA("Attachment") then
                                local OrigAtt = FindFirstChild(OriginalPart, Child.Name)
                                
                                if OrigAtt then
                                    Child.CFrame = OrigAtt.CFrame
                                end
                            end
                        end
                    end
                end
                
                local Head = FindFirstChild(PlayerCharacter, "Head")
                local LowerLeg = FindFirstChild(PlayerCharacter, "RightLowerLeg")
                local Foot = FindFirstChild(PlayerCharacter, "RightFoot")
                
                if Head and self.OriginalState.FaceTexture ~= nil then
                    local CharFace = FindFirstChildOfClass(Head, "Decal")
                    if self.OriginalState.FaceTexture ~= "" then
                        if CharFace then
                            CharFace.Texture = self.OriginalState.FaceTexture
                        else
                            local NewFace = Instance.new("Decal")
                            NewFace.Name = "face"
                            NewFace.Texture = self.OriginalState.FaceTexture
                            NewFace.Parent = Head
                        end
                    else
                        if CharFace then
                            CharFace:Destroy()
                        end
                    end
                end
                
                if Head and self.OriginalState.Transparencies.Head ~= nil then
                    Head.Transparency = self.OriginalState.Transparencies.Head
                    
                    for _, Descendant in GetDescendants(Head) do
                        if Descendant:IsA("Decal") or Descendant:IsA("Texture") then
                            Descendant.Transparency = self.OriginalState.Transparencies.Head
                        end
                    end
                end
                
                if LowerLeg and self.OriginalState.Transparencies.RightLowerLeg ~= nil then
                    LowerLeg.Transparency = self.OriginalState.Transparencies.RightLowerLeg
                    
                    for _, Child in GetChildren(LowerLeg) do
                        if Child:IsA("Decal") or Child:IsA("Texture") then
                            Child.Transparency = self.OriginalState.Transparencies.RightLowerLeg
                        end
                    end
                end
                
                if Foot and self.OriginalState.Transparencies.RightFoot ~= nil then
                    Foot.Transparency = self.OriginalState.Transparencies.RightFoot
                    
                    for _, Child in GetChildren(Foot) do
                        if Child:IsA("Decal") or Child:IsA("Texture") then
                            Child.Transparency = self.OriginalState.Transparencies.RightFoot
                        end
                    end
                end
                
                for _, Joint in GetDescendants(PlayerCharacter) do
                    if Joint:IsA("Motor6D") then
                        local Part0 = Joint.Part0
                        local Part1 = Joint.Part1
                        
                        if Part0 and Part1 then
                            local AttName = Joint.Name .. "RigAttachment"
                            local Att0 = FindFirstChild(Part0, AttName)
                            local Att1 = FindFirstChild(Part1, AttName)
                            
                            if Att0 and Att1 then
                                Joint.C0 = Att0.CFrame
                                Joint.C1 = Att1.CFrame
                            end
                        end
                    end
                end
            end
            
            for _, ClothingItem in self.OriginalState.Clothing do
                local ClonedClothing = ClothingItem:Clone()
                ClonedClothing.Parent = PlayerCharacter
            end
            
            if Humanoid then
                for _, AccessoryItem in self.OriginalState.Accessories do
                    local Clone = AccessoryItem:Clone()
                    local Handle = FindFirstChild(Clone, "Handle")
                    
                    if Handle then
                        for _, Weld in GetChildren(Handle) do
                            if Weld:IsA("JointInstance") then
                                Weld:Destroy()
                            end

        function AvatarSpoofer:Apply(PlayerCharacter, TargetUserId, ScalingMode)
            if not TargetUserId or TargetUserId == "" then
                return nil
            end

            local UserId = self:ResolveUserId(TargetUserId)
            
            if not UserId then
                return nil
            end

            local Description = self:FetchDescription(UserId)
            
            if not Description then
                return nil
            end

            local Preset = ScalePresets[ScalingMode]
            
            if Preset then
                for _, ScaleInfo in ScaleMap do
                    pcall(function()
                        Description[ScaleInfo[2]] = Preset[ScaleInfo[1]]
                    end)
                end
            end

            local Appearance = self:FetchAppearance(UserId)
            local Dummy = self:CreateDummy(Description)
            
            if Dummy then
                self:ApplyBodyParts(PlayerCharacter, Dummy)
                
                if Appearance then
                    self:ApplyClothingAndAccessories(PlayerCharacter, Appearance)
                    Appearance:Destroy()
                else
                    self:ApplyClothingAndAccessories(PlayerCharacter, Dummy)
                end
                
                self:ApplyBodyColors(PlayerCharacter, Dummy)
                self:ApplyFace(PlayerCharacter, Dummy)

                Dummy:Destroy()
            end
            
            return Description
        end

            end

            local CurrentCharacter = nil
            local IsFirstLoad = true

            local function AddCharacter(PlayerCharacter)
                CurrentCharacter = PlayerCharacter
                
                PlayerCharacter:WaitForChild("Humanoid")
                task.wait(0.5)

                local Config = shared.prada.Character["Avatar-Spoofer"]
                
                if not Config then
                    return
                end
                
                local Settings = { Avatar = Config.Avatar, Character = Config.Character, Animation = Config.Animation }
                local ShouldProcess = false
                
                if Config.Enabled then
                    local FeaturesEnabled = false
                    
                    if Settings['Avatar']['Enable'] then FeaturesEnabled = true end
                    if Settings['Character']['Headless'] then FeaturesEnabled = true end
                    if Settings['Character']['Korblox'] then FeaturesEnabled = true end
                    if Settings['Character']['Scaling'] ~= "" and Settings['Character']['Scaling'] ~= "Target" then FeaturesEnabled = true end

                    if FeaturesEnabled then
                        if IsFirstLoad then
                            ShouldProcess = true
                        elseif Settings['Avatar']['Apply Respawn'] then
                            ShouldProcess = true
                        end
                    end
                end

                IsFirstLoad = false

                if ShouldProcess then
                    AvatarSpoofer:Process(PlayerCharacter)
                end

                if AvatarSpoofer.StateLoop then
                    task.cancel(AvatarSpoofer.StateLoop)
                    AvatarSpoofer.StateLoop = nil
                end

                AvatarSpoofer.StateLoop = task.spawn(function()
                    local AppliedState = {
                        Headless = Settings['Character']['Headless'],
                        Korblox = Settings['Character']['Korblox'],
                        Scaling = Settings['Character']['Scaling'],
                        AvatarEnabled = Settings['Avatar']['Enable'],
                        UserId = Settings['Avatar']['UserId'],
                        ApplyRespawn = Settings['Avatar']['Apply Respawn']
                    }

                    while CurrentCharacter == PlayerCharacter and PlayerCharacter.Parent do
                        task.wait(0.1)
                        
                        local LiveConfig = shared.prada.Character["Avatar-Spoofer"]
                        
                        if not LiveConfig or not LiveConfig.Enabled then
                            continue
                        end
                        
                        local LiveSettings = { Avatar = LiveConfig.Avatar, Character = LiveConfig.Character }
                        local HasChanged = false
                        
                        if LiveSettings['Character']['Headless'] ~= AppliedState.Headless then
                            HasChanged = true
                        end
                        
                        if LiveSettings['Character']['Korblox'] ~= AppliedState.Korblox then
                            HasChanged = true
                        end
                        
                        if LiveSettings['Character']['Scaling'] ~= AppliedState.Scaling then
                            HasChanged = true
                        end
                        
                        if LiveSettings['Avatar']['Enable'] ~= AppliedState.AvatarEnabled then
                            HasChanged = true
                        end
                        
                        if LiveSettings['Avatar']['UserId'] ~= AppliedState.UserId then
                            HasChanged = true
                        end

                        if LiveSettings['Avatar']['Apply Respawn'] ~= AppliedState.ApplyRespawn then
                            HasChanged = true
                        end
                        
                        if HasChanged then
                            AppliedState.Headless = LiveSettings['Character']['Headless']
                            AppliedState.Korblox = LiveSettings['Character']['Korblox']
                            AppliedState.Scaling = LiveSettings['Character']['Scaling']
                            AppliedState.AvatarEnabled = LiveSettings['Avatar']['Enable']
                            AppliedState.UserId = LiveSettings['Avatar']['UserId']
                            AppliedState.ApplyRespawn = LiveSettings['Avatar']['Apply Respawn']
                            
                            AvatarSpoofer:Revert(PlayerCharacter)
                            AvatarSpoofer:Process(PlayerCharacter)
                        end
                    end
                end)
            end

            Utility:CreateConnections(LocalPlayer.CharacterAdded, function(PlayerCharacter)
                AddCharacter(PlayerCharacter)
            end)

            local PlayerCharacter = Character:GetCharacters(LocalPlayer)

            if PlayerCharacter then
                AddCharacter(PlayerCharacter)
            end
        end

        __Avatar_Handling()
    end

        function AnimationSpoofer:ResolveId(Input, AnimType)
            if not Input or Input == "" then
                return nil
            end
            
            if ANIM_PRESETS[Input] and ANIM_PRESETS[Input][AnimType] then
                return ANIM_PRESETS[Input][AnimType]
            end
            
            if string.find(Input, "rbxassetid://") then
                return Input
            end
            
            if tonumber(Input) then
                return "rbxassetid://" .. Input
            end
            
            return nil
        end

        function AnimationSpoofer:SaveOriginal(PlayerCharacter)
            if self.OriginalState then
                return

            local Animate = PlayerCharacter:FindFirstChild("Animate")

                return
            end
            
            self.OriginalState = {
                Idle1 = nil,
                Idle2 = nil,
                Walk = nil,
                Run = nil,
                Jump = nil,
                Fall = nil

            local IdleFolder = Animate:FindFirstChild("idle")

                local Anim = WalkFolder:FindFirstChild("WalkAnim")

        function AnimationSpoofer:Revert(PlayerCharacter)
            if not self.OriginalState then
                return
            end
            
            local Animate = PlayerCharacter:FindFirstChild("Animate")
            
            if not Animate then
                return
            end
            
            local IdleFolder = Animate:FindFirstChild("idle")
            
            if IdleFolder then
                local Anim1 = IdleFolder:FindFirstChild("Animation1")
                local Anim2 = IdleFolder:FindFirstChild("Animation2")
                
                if Anim1 and self.OriginalState.Idle1 then 
                    Anim1.AnimationId = self.OriginalState.Idle1 
                end
                
                if Anim2 and self.OriginalState.Idle2 then 
                    Anim2.AnimationId = self.OriginalState.Idle2 
                end
            end
            
            local WalkFolder = Animate:FindFirstChild("walk")
            
            if WalkFolder then
                local Anim = WalkFolder:FindFirstChild("WalkAnim")
                
                if Anim and self.OriginalState.Walk then 
                    Anim.AnimationId = self.OriginalState.Walk 
                end
            end
            
            local RunFolder = Animate:FindFirstChild("run")
            
            if RunFolder then
                local Anim = RunFolder:FindFirstChild("RunAnim")
                
                if Anim and self.OriginalState.Run then 
                    Anim.AnimationId = self.OriginalState.Run

            local JumpFolder = Animate:FindFirstChild("jump")

                local Anim = JumpFolder:FindFirstChild("JumpAnim")

                    Anim.AnimationId = self.OriginalState.Jump

            local FallFolder = Animate:FindFirstChild("fall")

                local Anim = FallFolder:FindFirstChild("FallAnim")

                    Anim.AnimationId = self.OriginalState.Fall

            self.OriginalState = nil
        end

        function AnimationSpoofer:Process(PlayerCharacter)
            local Config = shared.prada.Character["Avatar-Spoofer"]
            
            if not Config or not Config.Enabled then
                return
            end
            
            local Settings = { Avatar = Config.Avatar, Character = Config.Character, Animation = Config.Animation }
            
            if not Settings['Animation']['Enable'] then
                return
            end
            
            local Animate = PlayerCharacter:FindFirstChild("Animate")
            
            if not Animate then
                return
            end
            
            self:SaveOriginal(PlayerCharacter)
            
            local IdleId = self:ResolveId(Settings['Animation']['Idle'], 'Idle')
            
            if IdleId then
                local IdleFolder = Animate:FindFirstChild("idle")
                
                if IdleFolder then
                    local Anim1 = IdleFolder:FindFirstChild("Animation1")
                    local Anim2 = IdleFolder:FindFirstChild("Animation2")
                    
                    if Anim1 then 
                        Anim1.AnimationId = IdleId 
                    end
                    
                    if Anim2 then 
                        Anim2.AnimationId = IdleId 
                    end
                end
            end
            
            local WalkId = self:ResolveId(Settings['Animation']['Walk'], 'Walk')
            
            if WalkId then
                local WalkFolder = Animate:FindFirstChild("walk")
                
                if WalkFolder then
                    local Anim = WalkFolder:FindFirstChild("WalkAnim")
                    
                    if Anim then 
                        Anim.AnimationId = WalkId 
                    end
                end
            end
            
            local RunId = self:ResolveId(Settings['Animation']['Run'], 'Run')
            
            if RunId then
                local RunFolder = Animate:FindFirstChild("run")
                
                if RunFolder then
                    local Anim = RunFolder:FindFirstChild("RunAnim")
                    
                    if Anim then 
                        Anim.AnimationId = RunId 
                    end
                end
            end
            
            local JumpId = self:ResolveId(Settings['Animation']['Jump'], 'Jump')
            
            if JumpId then
                local JumpFolder = Animate:FindFirstChild("jump")
                
                if JumpFolder then
                    local Anim = JumpFolder:FindFirstChild("JumpAnim")
                    
                    if Anim then 
                        Anim.AnimationId = JumpId 
                    end
                end
            end
            
            local FallId = self:ResolveId(Settings['Animation']['Fall'], 'Fall')
            
            if FallId then
                local FallFolder = Animate:FindFirstChild("fall")
                
                if FallFolder then
                    local Anim = FallFolder:FindFirstChild("FallAnim")
                    
                    if Anim then 
                        Anim.AnimationId = FallId 
                    end
                end
            end
        end

        local function __Animation_Handling()
            local CurrentCharacter = nil

            local function AddCharacter(PlayerCharacter)
                CurrentCharacter = PlayerCharacter
                
                PlayerCharacter:WaitForChild("Animate")
                task.wait(0.5)

                local Config = shared.prada.Character["Avatar-Spoofer"]
                
                if not Config then 
                    return 
                end
                
                local Settings = { Avatar = Config.Avatar, Character = Config.Character, Animation = Config.Animation }
                
                if Config.Enabled and Settings['Animation']['Enable'] then
                    AnimationSpoofer:Process(PlayerCharacter)
                end

                if AnimationSpoofer.StateLoop then
                    task.cancel(AnimationSpoofer.StateLoop)
                    AnimationSpoofer.StateLoop = nil
                end

                AnimationSpoofer.StateLoop = TaskSpawn(function()
                    local LoopConfig = shared.prada.Character["Avatar-Spoofer"]
                    
                    if not LoopConfig then 
                        return 
                    end
                    
                    local LoopSettings = { Avatar = LoopConfig.Avatar, Character = LoopConfig.Character, Animation = LoopConfig.Animation }
                    
                    local AppliedState = {
                        Enable = LoopSettings['Animation']['Enable'],
                        Idle = LoopSettings['Animation']['Idle'],
                        Walk = LoopSettings['Animation']['Walk'],
                        Run = LoopSettings['Animation']['Run'],
                        Jump = LoopSettings['Animation']['Jump'],
                        Fall = LoopSettings['Animation']['Fall']
                    }

                    while CurrentCharacter == PlayerCharacter and PlayerCharacter.Parent do
                        task.wait(0.1)
                        
                        local LiveConfig = shared.prada.Character["Avatar-Spoofer"]
                        
                        if not LiveConfig or not LiveConfig.Enabled then
                            continue
                        end
                        
                        local LiveSettings = { Avatar = LiveConfig.Avatar, Character = LiveConfig.Character, Animation = LiveConfig.Animation }
                        local HasChanged = false
                        
                        if LiveSettings['Animation']['Enable'] ~= AppliedState.Enable then 
                            HasChanged = true 
                        end
                        
                        if LiveSettings['Animation']['Idle'] ~= AppliedState.Idle then 
                            HasChanged = true 
                        end
                        
                        if LiveSettings['Animation']['Walk'] ~= AppliedState.Walk then 
                            HasChanged = true 
                        end
                        
                        if LiveSettings['Animation']['Run'] ~= AppliedState.Run then 
                            HasChanged = true 
                        end
                        
                        if LiveSettings['Animation']['Jump'] ~= AppliedState.Jump then 
                            HasChanged = true 
                        end
                        
                        if LiveSettings['Animation']['Fall'] ~= AppliedState.Fall then 
                            HasChanged = true 
                        end
                        
                        if HasChanged then
                            AppliedState.Enable = LiveSettings['Animation']['Enable']
                            AppliedState.Idle = LiveSettings['Animation']['Idle']
                            AppliedState.Walk = LiveSettings['Animation']['Walk']
                            AppliedState.Run = LiveSettings['Animation']['Run']
                            AppliedState.Jump = LiveSettings['Animation']['Jump']
                            AppliedState.Fall = LiveSettings['Animation']['Fall']
                            
                            AnimationSpoofer:Revert(PlayerCharacter)
                            AnimationSpoofer:Process(PlayerCharacter)
                        end
                    end
                end)
            end

            Utility:CreateConnections(LocalPlayer.CharacterAdded, function(PlayerCharacter)
                AddCharacter(PlayerCharacter)
            end)

            local PlayerCharacter = Character:GetCharacters(LocalPlayer)

            if PlayerCharacter then
                AddCharacter(PlayerCharacter)
            end
        end

        __Animation_Handling()
    end
